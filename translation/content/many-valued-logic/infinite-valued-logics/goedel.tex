\olsection{గోడెల్ తర్కాలు}

\begin{editorial}
  ఇది అనంత-విలువల గోడెల్ తర్కంపై చిన్న ముసాయిదా విభాగం.
\end{editorial}

\begin{defn}\ollabel{def:goedel}
అనంత-విలువల గోడెల్
తర్కం~$\LogGod[\infty]$ను కింది మాత్రికతో
నిర్వచిస్తాం:
\begin{enumerate}
  \item $\lfalse$, $\lnot$, $\land$, $\lor$, $\lif$
  సంయోజకాలు గల ప్రామాణిక ప్రతిజ్ఞావాక్య
  భాష $\Lang L_0$.
  \item సత్యమూల్యాల సమితి $V_\infty$.
  \item $1$ ఒక్కటే నిర్దేశిత విలువ. అంటే
  $V^+ = \{1\}$.
  \item సత్యమూల్య ప్రమేయాలు ఇలా ఉంటాయి.
  \begin{align*}
    \tf{\lfalse} & = 0\\
    \tf{\lnot}[\LogGod](x) & = \begin{cases}
      1 & \text{ఒకవేళ } x =0\\
      0 & \text{లేకపోతే}
    \end{cases}\\
    \tf{\land}[\LogGod](x,y) & = \min(x,y)\\
    \tf{\lor}[\LogGod](x,y) & = \max(x,y)\\
    \tf{\lif}[\LogGod](x,y) & = \begin{cases}
      1 & \text{ఒకవేళ } x \le y\\
      y & \text{లేకపోతే.}
    \end{cases}
    \end{align*}
\end{enumerate}
$V = V_m$గా తీసుకుని, మిగతాదంతా ఇదే విధంగా ఉంచి $m$-విలువల
గోడెల్ తర్కాన్ని నిర్వచిస్తాం.
\end{defn}

\begin{prop}
  \olref[thr][god]{defn:goedel}లో
  నిర్వచించిన $\LogGod[3]$ తర్కమే
  \olref{def:goedel}లో
  నిర్వచించిన $\LogGod[3]$ తర్కం.
\end{prop}

\begin{proof}
  \olref[thr][god]{defn:goedel}లో
   సంయోజకాలకు ఇచ్చిన సత్య పట్టికలను
   \olref{def:goedel}లో ఇచ్చిన సమీకరణాల నుంచి వచ్చే
   కింది సత్య పట్టికలతో పోల్చితే రెండు నిర్వచనాలూ ఒకే తర్కాన్ని ఇస్తాయని తెలుస్తుంది.
  \begin{center}
    \begin{tabular}{c|c}
      $\tf{\lnot}[\LogGod[3]]$ & \\
      \hline
      $1$ & $0$ \\
      $1/2$ & $0$ \\
      $0$ & $1$
    \end{tabular}
    \quad
    \begin{tabular}{c|ccc}
      $\tf{\land}[\LogGod]$ & $1$ & $1/2$ & $0$ \\
      \hline
      $1$ & $1$ & $1/2$ & $0$ \\
      $1/2$ & $1/2$ & $1/2$ & $0$\\
      $0$ & $0$ & $0$ & $0$
    \end{tabular}
    \\[2ex]
    \begin{tabular}{c|ccc}
      $\tf{\lor}[\LogGod]$ & $1$ & $1/2$ & $0$ \\
      \hline
      $1$ & $1$ & $1$ & $1$ \\
      $1/2$ & $1$ & $1/2$ & $1/2$ \\
      $0$ & $1$ & $1/2$ & $0$
    \end{tabular}
    \quad
    \begin{tabular}{c|ccc}
      $\tf{\lif}[\LogGod]$ & $1$ & $1/2$ & $0$ \\
      \hline
      $1$ & $1$ & $1/2$ & $0$ \\
      $1/2$ & $1$ & $1$ & $0$  \\
      $0$ & $1$ & $1$ & $1$
    \end{tabular}
  \end{center}
\end{proof}

\begin{prop}\ollabel{prop:god-infty-m}
  $\Gamma \Entails[\LogGod[\infty]] !B$
  అయితే ప్రతి~$m \ge 2$కు
  $\Gamma \Entails[\LogGod[m]] !B$.
\end{prop}

\begin{proof}
  సాధన.
\end{proof}

\begin{prob}
  \olref[mvl][inf][god]{prop:god-infty-m}ను
  నిరూపించండి.
\end{prob}

పూర్వాధారాల సమితి పరిమితమైతే, తిరుగు వాదమూ నిజమే.

$\LogGod[3]$లాగే, $\LogGod[\infty]$లోనూ
అంతర్బోధవాద తర్కంలో చెల్లే
\tetoken{సూత్రాలు}{!!{formula}s}
అన్నీ సర్వసత్యాలే. మూడు-విలువల సందర్భంలో సర్వసత్యాలు కానివిగా
చూపిన కింది ఉదాహరణలు~$\LogGod[\infty]$లోనూ సర్వసత్యాలు కావు:
\begin{align*}
  & p \lor \lnot p && (p \lif q) \lif (\lnot p \lor q) \\
  & \lnot\lnot p \lif p && \lnot(\lnot p \land \lnot q) \lif (p \lor q) \\
  & ((p \lif q) \lif p) \lif p && \lnot(p \lif q) \lif (p \land \lnot q)
\end{align*}
అంతర్బోధవాద తర్కంలో చెల్లకపోయినా~$\LogGod[3]$లో
సర్వసత్యమైన \tetoken{సూత్రానికి}{!!{formula}} ఉదాహరణ
$(p \lif q) \lor (q \lif
p)$. అది~$\LogGod[\infty]$లోనూ సర్వసత్యమే.
అంతర్బోధవాద తర్కానికి $(!A \lif !B) \lor (!B \lif !A)$
పథకాన్ని చేరిస్తే సరిగ్గా $\LogGod[\infty]$ తర్కమే వస్తుందని
మైకేల్ డమ్మెట్ చూపించాడు. అందువల్ల $\LogGod[\infty]$ను
తరచుగా గోడెల్--డమ్మెట్ తర్కం~$\Log{LC}$ అంటారు.

\begin{prob}
  $(p \lif q) \lor (q \lif p)$
  ~$\LogGod[\infty]$లో సర్వసత్యమని
  చూపండి.
\end{prob}

\begin{prob}
  $\LogGod[3]$లో సర్వసత్యమైన
  $(p \lif q) \lor (q \lif r) \lor (r \lif s)$
  అనేది~$\LogGod[\infty]$లో
  సర్వసత్యం కాదని చూపండి.
\end{prob}
