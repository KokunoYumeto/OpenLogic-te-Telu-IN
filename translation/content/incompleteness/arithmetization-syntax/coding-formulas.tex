\olfileid{inc}{art}{frm}
\olsection{\printtoken{P}{formula} సంకేతీకరణ}

$\fn{Term}(x)$ సంబంధాన్ని ఆదిమ పునరావృత్తంగా నిర్వచించిన తరువాత, దాన్ని
ఉపయోగించి \tetoken{సూత్రాలకు}{!!{formula}s} సంబంధించిన $\fn{Frm}(x)$
సంబంధాన్ని ఆదిమ పునరావృత్తంగా నిర్వచించవచ్చు.

\begin{prop}
$x$ ఒక పరమాణు \tetoken{సూత్రానికి}{!!{formula}} గ్యోడెల్ సంఖ్య అయితేనే
నిలిచే $\fn{Atom}(x)$ సంబంధం ఆదిమ పునరావృత్తం.
\end{prop}

\begin{proof}
కింది షరతుల్లో కనీసం ఒకటి నెరవేరితే, అప్పుడే మాత్రమే $x$ ఒక పరమాణు
\tetoken{సూత్రానికి}{!!{formula}} గ్యోడెల్ సంఖ్య:
\begin{enumerate}
\item $\ell=\len{x}$, $B(x)=p_\ell^{(\ell+1)(x+1)}$ అనుకుందాం.
  $n,j<x$ అనుకుందాం. $\fn{Seq}(z)$,
  $\len{z}=n$, $z\leq B(x)$ నెరవేర్చే చెల్లుబాటు అయ్యే క్రమ సంకేతం $z$ ఒకటి ఉంది. ప్రతి $i<n$కి
  $\fn{Term}((z)_i)$ అవుతుంది. ఇంకా $x=$
\[
\Gn{\Obj P^n_j(} \concat \fn{flatten}(z) \concat \Gn{)}.
\]
\item $z_1,z_2<x$ ఉన్నాయి; $\fn{Term}(z_1)$, $\fn{Term}(z_2)$, ఇంకా
  $x=$
\[
\Gn{{\eq}(} \concat z_1 \concat \Gn{,} \concat z_2 \concat{\Gn{)}}.
\]
  \tagitem{prvFalse}{$x = \Gn{\lfalse}$.}{}
  \tagitem{prvTrue}{$x = \Gn{\ltrue}$.}{}
\end{enumerate}
\end{proof}

\begin{prop}
\ollabel{prop:frm-primrec}
$x$ ఒక \tetoken{సూత్రానికి}{!!a{formula}} గ్యోడెల్ సంఖ్య అయితే, అప్పుడే మాత్రమే వర్తించే
$\fn{Frm}(x)$ సంబంధం ఆదిమ పునరావృత్తం.
\end{prop}

\begin{proof}
సంకేతాల అనుక్రమం~$s$ ఒక \tetoken{సూత్రం}{!!a{formula}} కావడానికి అవసరమైన,
సరిపడే షరతు ఇది. పరమాణు \tetoken{సూత్రాల}{!!{formula}s} నుంచి నిర్మాణ
నియమాల ప్రకారం $s$ ఎలా ఏర్పడిందో నమోదు చేసే $s_0$, \dots,
$s_{k-1}=s$ అనే \tetoken{సూత్రాల}{!!{formula}} నిర్మాణ అనుక్రమం ఉండాలి.
సూత్రం~$s$కు సంకేతసంఖ్య~$x$ అనుకుందాం. సూత్రంలోని ఉపసూత్రాలతో నిర్మాణ
క్రమాన్ని ఎంచుకుంటే, ప్రతి $s_i$ సంకేతసంఖ్య $\leq x$ అవుతుంది. ఆ క్రమపు
పొడవు $k\leq\ell=\len{x}$. కాబట్టి నిర్మాణ క్రమం
$\tuple{s_0,\dots,s_{k-1}}$ సంకేతసంఖ్యను $p_{\ell-1}^{\ell(x+1)}$తో
పరిమితం చేయవచ్చు. పదాల విషయంలో చేసినట్టే, ఈ పరిమితిలో వెతికి, సంబంధాన్ని
పరిమిత పరిమాణీకరణతో ఆదిమ పునరావృత్తంగా నిర్వచించవచ్చు.
\end{proof}

\begin{prob}
\olref[inc][art][frm]{prop:frm-primrec}కు,
\olref[inc][art][trm]{prop:term-primrec} మొదటి నిరూపణ మాదిరిగా సవివరమైన
నిరూపణ ఇవ్వండి.
\end{prob}

\begin{prop}
\ollabel{prop:freeocc-primrec}
గ్యోడెల్ సంఖ్య~$x$ గల సూత్రంలో, గ్యోడెల్ సంఖ్య~$z$ గల చరం $i$వ సంకేతంగా
స్వేచ్ఛగా వస్తేనే $\fn{FreeOcc}(x,z,i)$ వర్తిస్తుంది. ఈ సంబంధం ఆదిమ పునరావృత్తం.
\end{prop}

\begin{proof}
అభ్యాసం.
\end{proof}

\begin{prob}
\olref[inc][art][frm]{prop:freeocc-primrec}ను నిరూపించండి. ఒక
\tetoken{సూత్రం}{!!a{formula}}లో వరుసగా ఉన్న సంకేతాల్లో కొన్నిటిని తీసుకుంటే
వచ్చే భాగం కూడా \tetoken{ఒక సూత్రమైతే}{!!a{formula}}, అది మొదటి సూత్రంలో
ఉప\tetoken{సూత్రం}{!!{formula}} అవుతుందనే విషయాన్ని ఉపయోగించవచ్చు.
\end{prob}

\begin{prop}
$x$ ఒక \tetoken{వాక్యానికి}{!!a{sentence}} గ్యోడెల్ సంఖ్య అయితే, అప్పుడే మాత్రమే వర్తించే
$\fn{Sent}(x)$ ధర్మం ఆదిమ పునరావృత్తం.
\end{prop}

\begin{proof}
\tetoken{వాక్యం}{!!{sentence}} అంటే \tetoken{చరాల}{!!{variable}s}
స్వేచ్ఛగా కనిపించని \tetoken{సూత్రం}{!!a{formula}}. కాబట్టి
$\fn{Sent}(x)$ వర్తించడానికి కింది షరతు అవసరం, సరిపోతుంది:
\begin{multline*}
\fn{Frm}(x) \land \bigl(\bforall{i<\len{x}}{\bforall{z<x}{}}\\
(\bexists{j<z}{z=\Gn{\Obj v_j}} \lif \lnot\fn{FreeOcc}(x,z,i))\bigr).
\end{multline*}
\end{proof}
