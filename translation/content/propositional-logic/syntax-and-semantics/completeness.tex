\olsection{ప్రతిజ్ఞావాక్య తర్కపు సంపూర్ణత}

\begin{defn}
\ollabel{def:MaxCon}
\tetoken{సూత్రాల}{!!{formula}s} సమితి $\Gamma$ సుసంగతమని అనుకుందాం. $\Gamma \subseteq \Delta$ అయ్యే ప్రతి సుసంగత సమితి $\Delta$కూ $\Gamma = \Delta$ అయితే, $\Gamma$ను \emph{గరిష్ఠ సుసంగత సమితి} అంటాం.
\end{defn}

\begin{lem}[సత్య ఉపసిద్ధాంతం]
  \ollabel{lem:truth}
$\Gamma$ గరిష్ఠ సుసంగత సమితి అనుకుందాం. అప్పుడు కింది మూడు వాక్యాలూ నిజం.
\begin{enumerate}
\item \ollabel{lem:truth-1} $!A \in \Gamma$ కావడానికి $\lnot !A \notin \Gamma$ అవసరం. ఆ షరతు సరిపోతుంది కూడా.
\item \ollabel{lem:truth-2} $\Gamma \Proves !A$ కావడానికి $!A \in \Gamma$ అవసరం. ఆ షరతు సరిపోతుంది కూడా.
\item \ollabel{lem:truth-3} $!A \lif !B \in \Gamma$ కావడానికి అవసరమైన, సరిపడే షరతు $!A \notin \Gamma$ లేదా $!B \in \Gamma$.
\end{enumerate}
\end{lem}

\begin{proof}
\begin{enumerate}
\item $!A \in \Gamma$, $\lnot !A \in \Gamma$ రెండూ అయితే $\Gamma$ అసంగతమవుతుంది. మరోవైపు, $!A$, $\lnot!A$ రెండూ $\Gamma$లో లేకపోతే, \olref[axd]{prop:phi} ప్రకారం $\Gamma\cup\{!A\}$, $\Gamma \cup \{\lnot !A\}$ రెండింటిలో కనీసం ఒకటి సుసంగతం. అప్పుడు $\Gamma$ గరిష్ఠమైనది కాదు.
\item $!A \in \Gamma$ అయితే $\Gamma \Proves !A$ అని స్పష్టమే. ఇక $\Gamma \Proves !A$ అనుకుందాం. $!A \notin \Gamma$ అయితే \olref{lem:truth-1} ప్రకారం $\lnot !A \in \Gamma$. అప్పుడు $\Gamma \Proves !A$, $\Gamma \Proves \lnot !A$ రెండూ వస్తాయి. ఇది వైరుధ్యం.
\item అభ్యాసం.
\end{enumerate}
\end{proof}

\olref{lem:truth}లోని \olref{lem:truth-2}లో ఎడమ నుంచి కుడికి ఉన్న దిశను తీసుకుంటే, $\Gamma$ \emph{నిగమనం పట్ల సంవృతమైనది} అని తెలుస్తుంది. అంటే, ఏ~$!A$కైనా $\Gamma \Proves !A$ అయితే $!A \in \Gamma$.

\begin{prob}
\olref{lem:truth} నిరూపణను పూర్తి చేయండి.
\end{prob}

\begin{thm}[సంపూర్ణత]
\ollabel{thm:completeness}
$\Gamma$ సుసంగతమైతే, $\Gamma$ సంతృప్తిపరచదగినది.
\end{thm}

\begin{proof}
భాషలోని \tetoken{సూత్రాలన్నింటినీ}{!!{formula}s} $!A_0$, $!A_1$,~\dots గా వరుసగా రాశామనుకుందాం. \tetoken{సూత్రాల}{!!{formula}s} సమితుల ఆరోహణ శ్రేణి $\Gamma_0$, $\Gamma_1$,~\dotsను పునరావర్తన పద్ధతిలో ఇలా నిర్వచిస్తాం.
\begin{align*}
\Gamma_0 & = \Gamma\\
\Gamma_{n+1} &  =
\begin{cases}
  \Gamma_n \cup \{!A_n\} & \text{$\Gamma_n \cup \{!A_n\}$ సుసంగతమైతే;}\\
  \Gamma_n \cup \{\lnot !A_n\} & \text{లేకపోతే}.
\end{cases}
\end{align*}
తర్వాత ఇలా నిర్వచిస్తాం:
\[
\Gamma^* = \bigcup_{0\le n}\Gamma_n.
\]
నిరూపణలో కింది విషయాలను వరుసగా నిరూపిస్తాం:
\begin{enumerate}
\item ప్రతి $n$కు $\Gamma_n$ సుసంగతం (\olref[axd]{prop:phi}ను ఉపయోగించి $n$పై ఆగమనంతో);
\item $\Gamma^*$ సుసంగతం;
\item $\Gamma^*$ గరిష్ఠమైనది.
\end{enumerate}
ఇప్పుడు $p_i \in \Gamma^*$ ఉన్నప్పుడే $v(p_i) = \True$ అయ్యేలా సత్యమూల్య కేటాయింపు $v$ను నిర్వచిస్తాం. $!A$పై ఆగమనంతో కింది విషయాన్ని చూపుతాం. మరింత సంక్లిష్టమైన \tetoken{వాక్యాల}{!!{sentence}s} విషయంలో కూడా $\Gamma^*$లో ఉండడమూ $v$ ప్రకారం సత్యం కావడమూ సమానార్థకాలు. అంటే $!A \in \Gamma^*$ ఉన్నప్పుడే $\overline{v}(!A) = \True$. ముఖ్యంగా, $v$ అనేది $\Gamma^*$ను సంతృప్తిపరుస్తుంది. $\Gamma \subseteq \Gamma^*$ కాబట్టి $\Gamma$ కూడా సంతృప్తిపరచదగినదే. మనం చూపవలసింది ఇదే.
\end{proof}

\begin{cor}
$\Gamma \Entails !A$ అయితే $\Gamma \Proves !A$.
\end{cor}

\begin{proof}
$\Gamma\Proves/ !A$ అయితే, \olref[axd]{prop:prov-incons} ప్రకారం $\Gamma \cup \{\lnot !A\}$ సుసంగతం. సంపూర్ణత సిద్ధాంతం ప్రకారం $\Gamma \cup \{\lnot !A\}$ సంతృప్తిపరచదగినది; అందువల్ల $\Gamma \Entails/ !A$.
\end{proof}

\begin{prop}[సంహతత్వ సిద్ధాంతం]
\ollabel{prop:compactness}
$\Gamma$ సంతృప్తిపరచదగినది కావడానికి అవసరమైన, సరిపడే షరతు $\Gamma_0$ అనే ప్రతి \emph{పరిమిత} ఉపసమితీ (అది $\Gamma$లోనిది) సంతృప్తిపరచదగినదిగా ఉండడం.
\end{prop}

\begin{proof}
$\Gamma$ అసంగతమైనప్పుడే అది సంతృప్తిపరచలేనిది. దానిలో అసంగతమైన పరిమిత ఉపసమితి $\Gamma_0$ ఒకటి ఉన్నప్పుడే అది అసంగతం. అలాంటి ఉపసమితి ఉండేది $\Gamma$లో సంతృప్తిపరచలేని పరిమిత ఉపసమితి $\Gamma_0$ ఒకటి ఉన్నప్పుడే.
\end{proof}

\begin{cor}
$\Gamma$లో $\Gamma_0 \Entails !A$ అయ్యే పరిమిత ఉపసమితి $\Gamma_0$ ఒకటి ఉండడం, $\Gamma \Entails !A$ కావడానికి అవసరమైన, సరిపడే షరతు.
\end{cor}
