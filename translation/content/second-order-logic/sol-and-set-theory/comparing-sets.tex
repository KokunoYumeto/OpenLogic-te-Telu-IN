\olsection{సమితులను పోల్చడం}

\begin{prop}
\tetoken{సూత్రం}{!!{formula}} $\lforall[x][(X(x) \lif Y(x))]$ ఉపసమితి
సంబంధాన్ని నిర్వచిస్తుంది. అంటే
$\Sat{M}{\lforall[x][(X(x) \lif Y(x))]}[s]$ నెరవేరితే, అప్పుడే మాత్రమే $s(X)
\subseteq s(Y)$.
\end{prop}

\begin{prop}
\tetoken{సూత్రం}{!!{formula}} $\lforall[x][(X(x) \liff Y(x))]$ సమితులపై
తాదాత్మ్య సంబంధాన్ని నిర్వచిస్తుంది. అంటే
$\Sat{M}{\lforall[x][(X(x) \liff Y(x))]}[s]$ నెరవేరితే, అప్పుడే మాత్రమే
$s(X) = s(Y)$.
\end{prop}

\begin{prop}
\tetoken{సూత్రం}{!!{formula}} $\lexists[x][X(x)]$ శూన్యేతరం అనే ధర్మాన్ని
నిర్వచిస్తుంది. అంటే $\Sat{M}{\lexists[x][X(x)]}[s]$ నెరవేరితే, అప్పుడే మాత్రమే
$s(X) \neq \emptyset$.
\end{prop}

సమితి~$X$ పరిమాణం సమితి~$Y$ పరిమాణాన్ని మించదు అని చెప్పడానికి
$\cardle{X}{Y}$ అని రాస్తాం. ఇది \tetoken{ఒకటి-ఒకటి}{!!a{injective}} ప్రమేయం
$f\colon X \to Y$ ఉన్నప్పుడే వర్తిస్తుంది. ఒక ప్రమేయం ఒకటి-ఒకటి
అనీ, అది~$X$లోని మూలకాలకు ఇచ్చే విలువలు~$Y$లో ఉంటాయనీ వ్యక్తం
చేయవచ్చు. కాబట్టి వ్యక్తి క్షేత్రంలోని ఉపసమితుల పరిమాణాలను
పోల్చే ఈ సంబంధాన్ని నిర్వచించవచ్చు.

\begin{prop}
ఈ సూత్రం
\[
\lexists[u][(\lforall[x][(X(x) \lif Y(u(x)))] \land \lforall[x][\lforall[y][(\eq[u(x)][u(y)] \lif
      \eq[x][y])]])]
\]
పరిమాణంలో మించని సంబంధాన్ని నిర్వచిస్తుంది.
\end{prop}

ఒకే పరిమాణం గల రెండు సమితులను “సమసంఖ్యాకాలు” అంటాం.
వాటికి $\cardeq{X}{Y}$ అని రాస్తాం. ఇది \tetoken{ద్విగుణ}{!!a{bijective}} ప్రమేయం
$f\colon X \to Y$ ఉన్నప్పుడే వర్తిస్తుంది.

\begin{prop}
ఈ సూత్రం
\begin{multline*}
  \lexists[u][(\lforall[x][(X(x) \lif Y(u(x)))] \land {}\\
    \lforall[x][\lforall[y][((X(x) \land X(y) \land
      \eq[u(x)][u(y)]) \lif \eq[x][y])]]
  \land {} \\\lforall[y][(Y(y) \lif \lexists[x][(X(x)
      \land \eq[y][u(x)])])])]
\end{multline*}
సమసంఖ్యాక సంబంధాన్ని నిర్వచిస్తుంది.
\end{prop}

ఈ \tetoken{సూత్రాలను}{!!{formula}s} వరుసగా $X \subseteq Y$,
$X = Y$, $X \neq \emptyset$, $\cardle{X}{Y}$, ఇంకా
$\cardeq{X}{Y}$ అని సంక్షిప్తంగా రాస్తాం. (సమితులు $X$, $Y$ గురించి
సాధారణంగా మాట్లాడేటప్పుడు కూడా ఇవే సంకేతాలు వాడతాం. అందువల్ల
కొంత గందరగోళం కలగవచ్చు. ఇక్కడ ఇవి ఏకస్థాన సంబంధ చరాలు
$X$,~$Y$ ఉన్న ద్వితీయ-స్థాయి తర్క
\tetoken{సూత్రాలకు}{!!{formula}s} సంక్షిప్త రూపం.)

\begin{prop}
\tetoken{వాక్యం}{!!{sentence}} $\lforall[X][\lforall[Y][((\cardle{X}{Y} \land
    \cardle{Y}{X}) \lif \cardeq{X}{Y})]]$ చెల్లుబాటవుతుంది.
\end{prop}

\begin{proof}
ఏ రెండు ఉపసమితులు $X \subseteq \Domain{M}$, $Y \subseteq \Domain{M}$ తీసుకున్నా,
$\cardle{X}{Y}$, $\cardle{Y}{X}$ అయితే $\cardeq{X}{Y}$ అయ్యే
పక్షంలో, \tetoken{వాక్యం}{!!{sentence}} \tetoken{ఒక నిర్మాణం}{!!a{structure}}~$\Struct{M}$లో
సంతృప్తమవుతుంది. అయితే ఇది \emph{ఏ} సమితులు $X$, $Y$కైనా
వర్తిస్తుంది. ఇదే ష్రోడర్--బెర్న్‌స్టైన్ సిద్ధాంతం.
\end{proof}
