\olsection{క్రమసంఖ్య కూడిక వినియోగం}

క్రమసంఖ్యల కూడికతో వివిధ సమితుల స్థాయిలను నేరుగా
లెక్కించవచ్చు. ఇక్కడ స్థాయి అంటే \olref[spine][rank]{defnsetrank}లో
నిర్వచించినదే.

\begin{lem}\ollabel{rankcomputation}
$\setrank{A} = \alpha$, $\setrank{B} = \beta$ అయితే:
\begin{enumerate}
	\item\ollabel{exrankpow} $\setrank{\Pow{A}} = \alpha\ordplus 1$
	\item\ollabel{exrankpair} $\setrank{\{A, B\}} = \max(\alpha,
	\beta) \ordplus 1$
	\item\ollabel{exrankcup} $\setrank{A \cup B} = \max(\alpha,
	\beta)$
	\item\ollabel{exranktuple} $\setrank{\tuple{A,B}} = \max(\alpha,
	\beta) \ordplus  2$
	\item\ollabel{exranktimes} $\setrank{A \times B} \leq \max(\alpha,
	\beta) \ordplus  2$
	\item\ollabel{exrankunion} $\alpha$ శూన్యం లేదా సీమా క్రమసంఖ్య అయితే $\setrank{\bigcup A} = \alpha$; $\alpha = \gamma\ordplus 1$ అయితే $\setrank{\bigcup A} = \gamma$
\end{enumerate}
\end{lem}

\begin{proof}
నిరూపణ అంతటా \olref[spine][rank]{ranksupstrict}ను పదేపదే వాడతాం.

\emph{\olref{exrankpow}.} $x \subseteq A$ అయితే $\setrank{x} \leq
\setrank{A}$. కాబట్టి $\setrank{\Pow{A}} \leq \alpha \ordplus  1$. ముఖ్యంగా $A
\in \Pow{A}$ కాబట్టి $\setrank{\Pow{A}} = \alpha \ordplus  1$.

\emph{\olref{exrankpair}.} \olref[spine][rank]{ranksupstrict} ప్రకారం.

\emph{\olref{exrankcup}.} \olref[spine][rank]{ranksupstrict} ప్రకారం.

\emph{\olref{exranktuple}.} \olref{exrankpair}ను రెండుసార్లు వాడితే.

\emph{\olref{exranktimes}.} $A \times B \subseteq
\Pow{\Pow{A \cup B}}$ అని గమనించి \olref{exranktuple}ను వాడితే ఫలితం వస్తుంది.

\emph{\olref{exrankunion}.} $\alpha = \gamma\ordplus 1$ అయితే $\setrank{c} = \gamma$ అయ్యే ఏదో $c \in A$ ఉంటుంది. $A$లోని ఏ \tetoken{మూలకం}{!!{element}} స్థాయీ అంతకన్నా పెద్దది కాదు. అందువల్ల $\setrank{\bigcup A} = \gamma$. $\alpha$ సీమా క్రమసంఖ్య అయితే $A$లో, $\alpha$కు ఎంత దగ్గరగా కావాలంటే అంత దగ్గరగా (కానీ దానికన్నా తక్కువగా) స్థాయి ఉన్న \tetoken{మూలకాలు}{!!{element}s} ఉంటాయి. అందువల్ల $\bigcup A$లో కూడా $\alpha$కు ఎంత దగ్గరగా కావాలంటే అంత దగ్గరగా (కానీ దానికన్నా తక్కువగా) స్థాయి ఉన్న \tetoken{మూలకాలు}{!!{element}s} ఉంటాయి. కాబట్టి $\setrank{\bigcup A} = \alpha$.
\end{proof}
\noindent
\olref{exranktimes}లో సమానత్వం కాక \emph{అసమానత్వం} ఎందుకు ఉందో చూపడం అభ్యాసానికి వదిలేస్తాం.

\begin{prob}
$\setrank{A \times B}=
\max(\setrank{A}, \setrank{B})$ అయ్యే సమితులు $A$, $B$లను కనుక్కోండి. అలాగే $\setrank{A \times B}=\max(\setrank{A}, \setrank{B}) \ordplus  2$ అయ్యే సమితులు $A$, $B$లను కనుక్కోండి. ఇతర స్థాయిలు సాధ్యమేనా?
\end{prob}

క్రమసంఖ్య పరిమితమా అనంతమా అని చెప్పడానికి సహజంగా తోచే
పలు ప్రమాణాలు తుల్యాలని ఇప్పుడు చూపవచ్చు.

\begin{lem}\ollabel{ordinfinitycharacter}
ఏ క్రమసంఖ్య $\alpha$కైనా ఈ కింది షరతులు తుల్యాలు:
\begin{enumerate}
	\item\ollabel{ord:notinomega} $\alpha\notin \omega$, అంటే $\alpha$ సహజ సంఖ్య కాదు
	\item\ollabel{ord:omegaplus} $\omega \leq \alpha$
	\item\ollabel{ord:oneplus} $1 \ordplus  \alpha = \alpha$
	%\alpha \approx \alpha \ordplus 1$
	\item\ollabel{ord:plusone} $\alpha \approx \alpha\ordplus 1$, అంటే $\alpha$, $\alpha\ordplus 1$ సమసంఖ్యాకాలు
	\item\ollabel{ord:infinite} $\alpha$ డెడెకిండ్ అనంతం
	\end{enumerate}
\end{lem}
\noindent
కాబట్టి ఒక క్రమసంఖ్య పరిమితమా అనంతమా అని చెప్పే ఐదు తుల్య ప్రమాణాలు మనకు లభించాయి.

\begin{proof}
\emph{\olref{ord:notinomega} $\Rightarrow$ \olref{ord:omegaplus}.} త్రివిధత ప్రకారం.

\emph{\olref{ord:omegaplus} $\Rightarrow$ \olref{ord:oneplus}.} $\alpha \geq \omega$ అనుకుందాం. అతిపరిమిత ఆగమనం ప్రకారం,
$\alpha = \beta \ordplus \gamma$ అయ్యే సీమా క్రమసంఖ్య
$\beta$ ఒకటి ఉండేలా చేసే అత్యల్ప క్రమసంఖ్య $\gamma$
ఉంటుంది (అది $0$ కూడా కావచ్చు). ఇప్పుడు:
\[
	1 \ordplus \alpha =
	1 \ordplus (\beta \ordplus \gamma) =
	(1 \ordplus \beta) \ordplus \gamma =
	\supstrict_{\delta < \beta} (1 \ordplus  \delta) \ordplus  \gamma =
	\beta \ordplus  \gamma =
	\alpha.
\]
\emph{\olref{ord:oneplus} $\Rightarrow$ \olref{ord:plusone}.} $f \colon (\alpha \disjointsum 1) \to (1
\disjointsum \alpha)$ అనే \tetoken{ద్విగుణ ప్రమేయం}{!!a{bijection}} స్పష్టంగా ఉంది. $1 \ordplus \alpha = \alpha$ అయితే $g \colon (1 \disjointsum \alpha) \to \alpha$ అనే సమరూపత ఉంటుంది. ఇప్పుడు $\comp{f}{g}$ను తీసుకుందాం.

\emph{\olref{ord:plusone} $\Rightarrow$ \olref{ord:infinite}.} $\alpha \approx \alpha \ordplus 1$ అయితే $f \colon (\alpha \disjointsum 1) \to \alpha$ అనే \tetoken{ద్విగుణ ప్రమేయం}{!!a{bijection}} ఉంటుంది. ప్రతి $\gamma < \alpha$కు $g(\gamma) = f(\gamma, 0)$గా నిర్వచిద్దాం. ఈ \tetoken{ఒకటి-ఒకటి ప్రమేయం}{!!{injection}} $f(0,1) \in \alpha \setminus \ran{g}$ అయినందున, $\alpha$ డెడెకిండ్ అనంతమని చూపుతుంది.

\emph{\olref{ord:infinite} $\Rightarrow$ \olref{ord:notinomega}.} ఇది \olref[z][infinity-again]{naturalnumbersarentinfinite}.
\end{proof}
