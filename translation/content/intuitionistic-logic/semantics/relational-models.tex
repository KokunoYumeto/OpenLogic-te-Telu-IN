\olsection{\usetoken{సంబంధ నమూనాలు}{relational model}}

అంతఃప్రజ్ఞావాద ప్రతిజ్ఞావాక్య తర్కానికి ఖచ్చితమైన
అర్థవిజ్ఞానం ఇవ్వాలంటే, ఏ నమూనాకు సంబంధించి
\tetoken{సూత్రాలను}{!!{formula}s} మూల్యాంకనం
చేయవచ్చో నిర్వచించాలి. అలాంటి నిర్వచనం
ఆధారంగా చెల్లుబాటు, అనుగమనం వంటి అర్థవిజ్ఞాన
భావనలను కూడా నిర్వచించవచ్చు. ఒక అర్థవిజ్ఞానాన్ని
\tetoken{సంబంధ నమూనాలు}{!!{relational model}s}
ఇస్తాయి.

\begin{defn}
  అంతఃప్రజ్ఞావాద ప్రతిజ్ఞావాక్య తర్కానికి
  \tetoken{ఒక సంబంధ నమూనా}{!!^a{relational model}}
  త్రయం $\mModel{M} = \tuple{W, R, V}$; ఇందులో
  \begin{enumerate}
  \item $W$ అనేది శూన్యం కాని సమితి,
  \item $R$ అనేది~$W$పై పాక్షిక క్రమం
    (అంటే స్వావర్తన, ప్రతిసౌష్ఠవ, సంక్రామక
    ద్విస్థానిక సంబంధం), మరియు
  \item $V$ అనేది ప్రతి \tetoken{ప్రతిజ్ఞావాక్య
    చరానికి}{!!{propositional
    variable}}~$p$కు~$W$ యొక్క ఉపసమితిని కేటాయించే
    ప్రమేయకం; దీనికి కింది షరతు ఉంటుంది:
  \item $V$ అనేది $R$కు సంబంధించి ఏకదిశగా పెరుగుతుంది;
    అంటే $w
    \in V(p)$ మరియు $Rww'$ అయితే $w' \in V(p)$.
  \end{enumerate}
\end{defn}

\begin{defn}\ollabel{defn:true-at-w}
  $\mModel{M}$లో $!A$ అనేది \emph{$w$ వద్ద సత్యం}
  కావడం అనే భావనను, $\mSat{M}{!A}[w]$ అనే సంకేతంతో,
  ఇలా ఆగమన పద్ధతిలో నిర్వచిస్తాం:
  \begin{enumerate}
  \item \indcase{!A}{p}{$\mSat{M}{\indfrm}[w]$ అయితే, అప్పుడే $w \in V(p)$.}
  \item \indcase{!A}{\lfalse}{$\mSat{M}{\indfrm}[w]$ కాదు}.
  \item \indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ అయితే, అప్పుడే
    $Rww'$ అయ్యే ఏ $w'$ వద్దా $\mSat{M}{!B}[w']$ కాదు}.
  \item \indcase{!A}{!B \land !C}{$\mSat{M}{\indfrm}[w]$ అయితే, అప్పుడే
    $\mSat{M}{!B}[w]$ మరియు $\mSat{M}{!C}[w]$}.
  \item \indcase{!A}{!B \lor !C}{$\mSat{M}{\indfrm}[w]$ అయితే, అప్పుడే
    $\mSat{M}{!B}[w]$ లేదా $\mSat{M}{!C}[w]$ (లేదా రెండూ)}.
  \item \indcase{!A}{!B \lif !C}{$\mSat{M}{\indfrm}[w]$ అయితే, అప్పుడే
    $Rww'$ అయ్యే ప్రతి $w'$ వద్ద $\mSat{M}{!B}[w']$ కాదు లేదా
    $\mSat{M}{!C}[w']$} (లేదా రెండూ).
  \end{enumerate}
  $\mSat{M}{!A}[w]$ కాకపోతే $\mSat/{M}{!A}[w]$
అని వ్రాస్తాం. $\Gamma$ అనేది
\tetoken{సూత్రాల}{!!{formula}s} సమితి అయితే,
$\mSat{M}{\Gamma}[w]$ అంటే ప్రతి $!B \in \Gamma$కు
$\mSat{M}{!B}[w]$ అని అర్థం.
\end{defn}

\begin{prob}
  \olref[int][sem][rel]{defn:true-at-w} ప్రకారం
  $\mSat{M}{\lnot !A}[w]$ అయితే, అప్పుడే
  $\mSat{M}{!A \lif \lfalse}[w]$ అని చూపండి.
\end{prob}

\begin{prop}\ollabel{prop:true-monotonic}
  లోకాల వద్ద సత్యం~$R$కు సంబంధించి ఏకదిశగా
  పెరుగుతుంది; అంటే $\mSat{M}{!A}[w]$ మరియు
  $Rww'$ అయితే $\mSat{M}{!A}[w']$.
\end{prop}

\begin{proof}
  అభ్యాసం.
\end{proof}

\begin{prob}
  \olref[int][sem][rel]{prop:true-monotonic}ను నిరూపించండి.
\end{prob}
