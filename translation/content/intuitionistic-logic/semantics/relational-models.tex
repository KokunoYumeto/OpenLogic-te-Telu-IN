\olsection{\usetoken{సంబంధ నమూనాలు}{relational model}}

అంతఃప్రజ్ఞావాద ప్రతిజ్ఞావాక్య తర్కానికి కచ్చితమైన
అర్థవిచారం ఇవ్వాలంటే, \tetoken{సూత్రాల}{!!{formula}s} సత్యాన్ని నిర్ణయించడానికి
ఎలాంటి నిర్మాణాన్ని నమూనాగా తీసుకుంటామో నిర్వచించాలి.
ఆ నిర్వచనం ఆధారంగా చెల్లుబాటు, అనుగమనం వంటి అర్థవిచార
భావనలనూ నిర్వచించవచ్చు. \tetoken{సంబంధ నమూనాలతో}{!!{relational model}s}
అలాంటి అర్థవిచారాన్ని ఇవ్వవచ్చు.

\begin{defn}
  అంతఃప్రజ్ఞావాద ప్రతిజ్ఞావాక్య తర్కానికి
  \tetoken{ఒక సంబంధ నమూనా}{!!^a{relational model}}
  త్రయం $\mModel{M} = \tuple{W, R, V}$; ఇందులో
  \begin{enumerate}
  \item $W$ అనేది శూన్యం కాని సమితి,
  \item $R$ అనేది~$W$పై పాక్షిక క్రమం
    (అంటే స్వావర్తన, ప్రతిసౌష్ఠవ, సంక్రామక ధర్మాలున్న
    ద్విస్థాన సంబంధం),
  \item $V$ అనేది ప్రతి \tetoken{ప్రతిజ్ఞావాక్య
    చరానికి}{!!{propositional
    variable}}~$p$కు~$W$లోని ఒక ఉపసమితిని కేటాయించే
    ప్రమేయం. దీనికి కింది షరతు వర్తిస్తుంది.
  \item $V$ అనేది $R$కు సంబంధించి ఏకదిశంగా ఉంటుంది.
    అంటే $w
    \in V(p)$ మరియు $Rww'$ అయితే $w' \in V(p)$.
  \end{enumerate}
\end{defn}

\begin{defn}\ollabel{defn:true-at-w}
  $\mModel{M}$లో $!A$ అనేది \emph{$w$ వద్ద సత్యం}
  అనే భావనను $\mSat{M}{!A}[w]$తో సూచిస్తాం.
  దీన్ని ఆగమనం ద్వారా ఇలా నిర్వచిస్తాం.
  \begin{enumerate}
  \item \indcase{!A}{p}{$\mSat{M}{\indfrm}[w]$ నెరవేరేది $w \in V(p)$ అయినప్పుడు, అప్పుడే మాత్రమే.}
  \item \indcase{!A}{\lfalse}{$\mSat{M}{\indfrm}[w]$ కాదు}.
  \item \indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ నెరవేరడానికి కింది షరతు అవసరం. అది ఉంటే సరిపోతుంది.
    $Rww'$ ఉన్న ఏ $w'$లోనూ $\mSat{M}{!B}[w']$ నెరవేరదు}.
  \item \indcase{!A}{!B \land !C}{$\mSat{M}{\indfrm}[w]$ అప్పుడే, అప్పుడు మాత్రమే
    $\mSat{M}{!B}[w]$ మరియు $\mSat{M}{!C}[w]$}.
  \item \indcase{!A}{!B \lor !C}{$\mSat{M}{\indfrm}[w]$ అప్పుడే, అప్పుడు మాత్రమే
    $\mSat{M}{!B}[w]$ లేదా $\mSat{M}{!C}[w]$ (లేదా రెండూ)}.
  \item \indcase{!A}{!B \lif !C}{$\mSat{M}{\indfrm}[w]$ వర్తించడానికి అవసరమైన, సరిపడే షరతు ఇది.
    $Rww'$ ఉన్న ప్రతి $w'$ వద్దా $\mSat{M}{!B}[w']$ వర్తించదు లేదా
    $\mSat{M}{!C}[w']$ వర్తిస్తుంది} (లేదా రెండూ).
  \end{enumerate}
  $\mSat{M}{!A}[w]$ నెరవేరకపోతే $\mSat/{M}{!A}[w]$
  అని రాస్తాం. $\Gamma$ అనేది
\tetoken{సూత్రాల}{!!{formula}s} సమితి అయితే,
$\mSat{M}{\Gamma}[w]$ అంటే ప్రతి $!B \in \Gamma$కు
$\mSat{M}{!B}[w]$ అని అర్థం.
\end{defn}

\begin{prob}
  \olref[int][sem][rel]{defn:true-at-w} ప్రకారం
  $\mSat{M}{\lnot !A}[w]$ అప్పుడే, అప్పుడు మాత్రమే
  $\mSat{M}{!A \lif \lfalse}[w]$ అని చూపండి.
\end{prob}

\begin{prop}\ollabel{prop:true-monotonic}
  లోకాల్లో సత్యం~$R$కు సంబంధించి ఏకదిశంగా ఉంటుంది.
  అంటే $\mSat{M}{!A}[w]$ నెరవేరి $Rww'$ అయితే,
  $\mSat{M}{!A}[w']$ కూడా నెరవేరుతుంది.
\end{prop}

\begin{proof}
  అభ్యాసం.
\end{proof}

\begin{prob}
  \olref[int][sem][rel]{prop:true-monotonic}ను నిరూపించండి.
\end{prob}
