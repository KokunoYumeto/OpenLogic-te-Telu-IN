\olsection{అర్థవిచార భావనలు}

\begin{defn}
  ప్రతి $w \in W$కు $\mSat{M}{!A}[w]$ నెరవేరినప్పుడు, అప్పుడే మాత్రమే
  $!A$ అనేది \emph{నమూనా~$\mModel{M} = \tuple{W,R,V}$లో
  సత్యం} అని అంటాం. దీన్ని $\mSat{M}{!A}$ అని రాస్తాం.
  అన్ని నమూనాల్లో సత్యమైనప్పుడు, అప్పుడే మాత్రమే $!A$ను
  \emph{చెల్లుబాటైనది} అంటాం. దీన్ని $\Entails !A$ అని రాస్తాం.
  ప్రతి నమూనా~$\mModel{M}$లోని ప్రతి~$w$లో
  $\mSat{M}{\Gamma}[w]$ నెరవేరితే $\mSat{M}{!A}[w]$
  కూడా నెరవేరాలి. ఈ షరతు నెరవేరినప్పుడు, అప్పుడే మాత్రమే
  \tetoken{సూత్రాల}{!!{formula}s} సమితి~$\Gamma$
  నుంచి~$!A$ \emph{అనుగమిస్తుంది} అని అంటాం.
  దీన్ని $\Gamma
  \Entails !A$ అని రాస్తాం.
\end{defn}

\begin{prop}\ollabel{prop:sat-entails}
  \begin{enumerate}
  \item\ollabel{prop:sat-entails1} $\mSat{M}{\Gamma}[w]$ మరియు
    $\Gamma \Entails !A$ అయితే $\mSat{M}{!A}[w]$.
  \item\ollabel{prop:sat-entails2} $\mSat{M}{\Gamma}$ మరియు $\Gamma
    \Entails !A$ అయితే $\mSat{M}{!A}$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item $\mSat{M}{\Gamma}[w]$ అనుకుందాం.
    $\Gamma \Entails !A$ కాబట్టి, అనుగమన
    నిర్వచనం ప్రకారం $\mSat{M}{!A}[w]$.
  \item \olref{prop:sat-entails1} నుంచి నేరుగా వస్తుంది.
  \end{enumerate}
\end{proof}


\begin{defn}\ollabel{defn:restrict}
  $\mModel{M}$ ఒక సంబంధ నమూనా,
  $w \in W$ అనుకుందాం. $w$కు సంబంధించి
  $\mModel{M}$ను \emph{పరిమితం} చేస్తే వచ్చే నమూనా
  $\mModel{M}_w=\tuple{W_w, R_w, V_w}$ను ఇలా నిర్వచిస్తాం.
  \begin{align*}
    W_w & = \Setabs{u \in W}{Rwu},\\
    R_w  & = R \cap (W_w)^2, \text{ మరియు}\\
    V_w(p) & = V(p) \cap W_w.
  \end{align*}
\end{defn}

\begin{prop}\ollabel{prop:restrict}
  $\mSat{M}{!A}[w]$ అనేది $\mSat{M_w}{!A}$కు తుల్యం.
\end{prop}

\begin{prob}
  \olref[int][sem][sem]{prop:restrict}ను నిరూపించండి.
\end{prob}

\begin{prop}
  $\mSat{M}{\Gamma}$ అయ్యే ప్రతి నమూనా~$\mModel{M}$లో
  $\mSat{M}{!A}$ అని అనుకుందాం. అప్పుడు
  $\Gamma \Entails !A$.
\end{prop}

\begin{proof}
  $\mSat{M}{\Gamma}[w]$ అనుకుందాం. ప్రతి
  $!B \in \Gamma$కు \olref{prop:restrict}ను
  వర్తింపజేస్తే $\mSat{M_w}{\Gamma}$ వస్తుంది.
  పరికల్పన ప్రకారం $\mSat{M_w}{!A}$.
  మళ్లీ \olref{prop:restrict} ద్వారా
  $\mSat{M}{!A}[w]$ వస్తుంది.
\end{proof}
