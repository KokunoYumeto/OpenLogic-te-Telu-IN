\olsection{స్వేచ్ఛా \printtoken{P}{variable}, \printtoken{P}{sentence}}

\begin{defn}[\tetoken{ఒక చరం}{!!a{variable}} స్వేచ్ఛగా కనిపించే స్థానాలు]
\ollabel{defn:free-occ}
\tetoken{ఒక సూత్రంలో}{!!a{formula}} \tetoken{చరం}{!!a{variable}} \emph{స్వేచ్ఛగా} కనిపించే స్థానాలను కింది
విధంగా ఆగమనాత్మకంగా నిర్వచిస్తాం:
\begin{enumerate}
\item \indcase*{!A}{$!A$ పరమాణువు}{ $\indfrm$లో \tetoken{చరం}{!!{variable}}
  కనిపించే ప్రతి స్థానంలోనూ ఆ చరం స్వేచ్ఛగా ఉంటుంది.}

\tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\indfrm$లో
  \tetoken{చరం}{!!{variable}} స్వేచ్ఛగా కనిపించే స్థానాలు $!B$లో అది స్వేచ్ఛగా కనిపించే స్థానాలే.}}{}

\item \indcase{!A}{(!B \ast !C)}{$\indfrm$లో
  \tetoken{చరం}{!!{variable}} స్వేచ్ఛగా కనిపించే స్థానాలు $!B$లో, $!C$లో అది స్వేచ్ఛగా కనిపించే స్థానాలే.}

\tagitem{prvAll}{\indcase{!A}{\lforall[x][!B]}{$\indfrm$లో
  \tetoken{చరం}{!!{variable}} స్వేచ్ఛగా కనిపించే స్థానాలు $!B$లో స్వేచ్ఛగా కనిపించే స్థానాలే. అయితే~$x$ కనిపించే స్థానాలను మినహాయించాలి.}}{}

\tagitem{prvEx}{\indcase{!A}{\lexists[x][!B]}{$\indfrm$లో
  \tetoken{చరం}{!!{variable}} స్వేచ్ఛగా కనిపించే స్థానాలు $!B$లో స్వేచ్ఛగా కనిపించే స్థానాలే. అయితే~$x$ కనిపించే స్థానాలను మినహాయించాలి.}}{}
\end{enumerate}
\end{defn}

\begin{defn}[బద్ధ చరాలు]
ఒక సూత్రం~$!A$లో \tetoken{ఒక చరం}{!!a{variable}} కనిపించే స్థానంలో అది స్వేచ్ఛగా లేకపోతే, ఆ స్థానంలో అది
\emph{బద్ధమైనది}.
\end{defn}

\begin{prob}
\olref[fol][syn][fvs]{defn:free-occ}ను అనుసరించి, చరాలు
బద్ధంగా కనిపించే స్థానాలకు ఆగమనాత్మక నిర్వచనం ఇవ్వండి.
\end{prob}

\begin{defn}[పరిధి]
\iftag{prvAll}{ఒక సూత్రం~$!A$లో $\lforall[x][!B]$ ఉపసూత్రంగా కనిపిస్తే,
  $!A$లో దానికి అనుగుణంగా కనిపించే~$!B$ను అక్కడి
  $\lforall[x]$ సంకేతానికి \emph{పరిధి} అంటాం. \iftag{prvEx}{అదే విధంగా
  $\lexists[x]$కూ.}{}}{ఒక సూత్రం~$!A$లో $\lexists[x][!B]$
  ఉపసూత్రంగా కనిపిస్తే, $!A$లో దానికి అనుగుణంగా కనిపించే~$!B$ను
  అక్కడి $\lexists[x]$ సంకేతానికి \emph{పరిధి} అంటాం.}

$!A$లో ఒక నిర్దిష్ట స్థానంలో కనిపించే పరిమాణీకరణికం
\iftag{prvAll}{$\lforall[x]$\iftag{prvEx}{ లేదా
    $\lexists[x]$}{}}{$\lexists[x]$}కు~$!B$ పరిధి అయితే, $!B$లో $x$
స్వేచ్ఛగా కనిపించే స్థానాల్లో అది
\iftag{prvAll}{$\lforall[x][!B]$లో\iftag{prvEx}{నూ
    $\lexists[x][!B]$లోనూ}{}}{$\lexists[x][!B]$లో} బద్ధమవుతుంది. ఈ
స్థానాల్లో చరాన్ని పైన పేర్కొన్న స్థానంలో ఉన్న పరిమాణీకరణికం \emph{బంధిస్తుంది} అంటాం.
\end{defn}

\begin{ex}
కింది సూత్రాన్ని పరిశీలించండి:
\[
\lexists[\Obj v_0][\underbrace{\Atom{\Obj A^2_0}{\Obj v_0,\Obj v_1}}_{!B}]
\]
$!B$, $\lexists[\Obj v_0]$ యొక్క పరిధిని సూచిస్తుంది.
పరిమాణీకరణికం $!B$లో కనిపించే $\Obj v_0$ను బంధిస్తుంది. అయితే
$\Obj v_1$ను బంధించదు. అందువల్ల ఈ సందర్భంలో $\Obj v_1$ స్వేచ్ఛా
చరం.


ఇది మరింత సంక్లిష్టమైన \tetoken{సూత్రం}{!!{formula}}~$!A$లో ఎలా పనిచేస్తుందో ఇప్పుడు
చూడవచ్చు:
\[
\lforall[\Obj v_0][\underbrace{(\Atom{\Obj A^1_0}{\Obj v_0} \lif
    \Atom{\Obj A^2_0}{\Obj v_0, \Obj v_1})}_{!B}] \lif \lexists[\Obj
  v_1][\underbrace{(\Atom{\Obj A^2_1}{\Obj v_0, \Obj v_1} \lor \lforall[\Obj v_0][\overbrace{\lnot \Atom{\Obj A^1_1}{\Obj v_0}}^{!D}])}_{!C}]
\]
$!B$, మొదటి $\lforall[\Obj v_0]$ పరిధి; $!C$, $\lexists[\Obj v_1]$
పరిధి; $!D$, రెండవ $\lforall[\Obj v_0]$ పరిధి. మొదటి
$\lforall[\Obj v_0]$,~$!B$లో $\Obj v_0$ కనిపించే అన్ని స్థానాల్లో ఆ చరాన్ని బంధిస్తుంది.
$\lexists[\Obj v_1]$,~$!C$లో కనిపించే $\Obj v_1$ను బంధిస్తుంది. రెండవ
$\lforall[\Obj v_0]$,~$!D$లో కనిపించే $\Obj v_0$ను బంధిస్తుంది. $!A$లో
మొదట కనిపించే $\Obj v_1$, నాలుగోసారి కనిపించే $\Obj v_0$ స్వేచ్ఛగా ఉంటాయి. చివరిసారి కనిపించే
$\Obj v_0$ అనేది $!D$లో స్వేచ్ఛగా ఉంటుంది. అయితే $!C$,~$!A$లలో బద్ధంగా ఉంటుంది.
\end{ex}

\begin{defn}[వాక్యం]
\tetoken{ఒక సూత్రం}{!!^a{formula}}~$!A$లో \tetoken{చరాలు}{!!{variable}s} ఎక్కడా స్వేచ్ఛగా కనిపించకపోతే,
అప్పుడు మాత్రమే అది \article{\tetoken{వాక్యం}{!!{sentence}}}.
\end{defn}
