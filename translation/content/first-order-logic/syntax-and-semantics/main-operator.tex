\olsection{ఒక సూత్రపు \printtoken{S}{main operator}}

\begin{explain}
\tetoken{ఒక సూత్రం}{!!a{formula}}~$!A$ను నిర్మించడంలో చివరిగా ఉపయోగించిన తార్కిక సంకేతం గురించి
మాట్లాడడం తరచుగా ఉపయోగకరం. ఆ సంకేతాన్ని $!A$కు
\emph{ప్రధాన తార్కిక సంకేతం} అంటాం. సులువుగా చెప్పాలంటే, ఇది $!A$లో
``అన్నిటికంటే బయట ఉన్న'' తార్కిక సంకేతం. ఉదాహరణకు, $\lnot !A$కు ప్రధాన తార్కిక సంకేతం
$\lnot$; $(!A \lor !B)$కు ప్రధాన తార్కిక సంకేతం $\lor$; ఇలాగే కొనసాగుతుంది.
\end{explain}


\begin{defn}[\tetoken{ప్రధాన తార్కిక సంకేతం}{!!^{main operator}}]
\ollabel{def:main-op}
\tetoken{ఒక సూత్రం}{!!a{formula}}~$!A$కు ఉన్న \emph{\tetoken{ప్రధాన తార్కిక సంకేతాన్ని}{!!{main operator}}} కింది విధంగా
నిర్వచిస్తాం:
\begin{enumerate}
\item \indcase*{!A}{!A}{$\indfrm$కు \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} ఉండదు.}

\tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\indfrm$ యొక్క \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}}
  $\lnot$.}}{}

\tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\indfrm$ యొక్క
  \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} $\land$.}}{}

\tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\indfrm$ యొక్క
  \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} $\lor$.}}{}

\tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\indfrm$ యొక్క
  \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} $\lif$.}}{}

\tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\indfrm$ యొక్క
  \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} $\liff$.}}{}

\tagitem{prvAll}{\indcase{!A}{\lforall[x][!B]}{$\indfrm$ యొక్క
  \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} $\lforall$.}}{}

\tagitem{prvEx}{\indcase{!A}{\lexists[x][!B]}{$\indfrm$ యొక్క
  \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} $\lexists$.}}{}
\end{enumerate}
\end{defn}

ప్రతి సందర్భంలోనూ \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} సూత్రంలో కనిపించే
నిర్దిష్ట \emph{స్థానాన్నే} ఉద్దేశిస్తున్నాం. ఉదాహరణకు,
$((!D \lif !E) \lif (!E \lif !D))$ సూత్రం $(!B \lif !C)$ రూపంలో
ఉంటుంది; ఇక్కడ $!B$ అనేది $(!D \lif !E)$, $!C$ అనేది
$(!E \lif !D)$. కనుక రెండవసారి కనిపించే $\lif$యే \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}}.

\begin{explain}
ఇది పరమాణుయేతర \tetoken{సూత్రాలు}{!!{formula}s} అన్నింటికీ వాటి \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} కనిపించే స్థానాన్ని
ఇచ్చే ప్రమేయానికి \emph{పునరావర్తక} నిర్వచనం. \tetoken{సూత్రాలను}{!!{formula}s}
ఆగమనాత్మకంగా నిర్వచించిన తీరు వల్ల ప్రతి \tetoken{సూత్రం}{!!{formula}}~$!A$,
\olref{def:main-op}లోని సందర్భాల్లో ఒకదాన్ని తీరుస్తుంది. అందువల్ల
ప్రతి పరమాణుయేతర \tetoken{సూత్రం}{!!{formula}}~$!A$కు \tetoken{ఒక ప్రధాన తార్కిక సంకేతం}{!!a{main
  operator}} ఉంటుందని హామీ.
ప్రతి \tetoken{సూత్రం}{!!{formula}} ఈ షరతుల్లో ఒక్కదాన్నే తీరుస్తుంది; ప్రతి సందర్భంలో
$!A$ను నిర్మించిన చిన్న \tetoken{సూత్రాలు}{!!{formula}s} ఒక్కొక్కటీ ఏకైకంగా నిర్ణయమవుతాయి. కనుక
$!A$లో \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main
  operator}} కనిపించే స్థానం ఒక్కటే. అందువల్ల మనం ఒక ప్రమేయాన్ని
నిర్వచించాం.
\end{explain}

ఒక సూత్రపు \tetoken{ప్రధాన తార్కిక సంకేతం}{!!{main operator}} ఏ సంకేతమో బట్టి, \olref{tab:main-op}లోని
పేర్లతో \tetoken{సూత్రాలను}{!!{formula}s} పిలుస్తాం.\iftag{defNot,defOr,defAnd,defIf,defIff,defTrue,defFalse,defEx,defAll}
{ అయితే నిర్వచిత తార్కిక సంకేతాలు అధికారికంగా \tetoken{సూత్రాలలో}{!!{formula}s} కనిపించవని
గుర్తుంచుకోండి. అవి కేవలం సంక్షిప్త రూపాలు. అందువల్ల అధికారికంగా అవి
ఒక సూత్రానికి ప్రధాన తార్కిక సంకేతాలు కాలేవు. కాబట్టి అన్ని \tetoken{సూత్రాలు}{!!{formula}s}
గురించిన నిరూపణల్లో వాటిని విడిగా పరిశీలించాల్సిన అవసరం లేదు.}{}
