\olsection{\usetoken{S}{derivability}, పరిమాణీకరణలు}

\begin{explain}
  సంపూర్ణతా సిద్ధాంతానికి మరో అవసరం ఉంది. ఈ విభాగంలో $\Proves$ గురించి
  నిరూపించే ఫలితాలు సహజ నిగమన నియమాల నుంచి రావాలి.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} $\Gamma$లో గానీ $!A(x)$లో గానీ
కనిపించని స్థిరసంకేతం $c$ అయి, $\Gamma \Proves !A(c)$ అయితే,
$\Gamma \Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
$\Gamma$ నుంచి $!A(c)$కు ఒక \tetoken{వ్యుత్పత్తిని}{!!a{derivation}} $\delta$ అందాం.
ఒక \Intro{\lforall} నిగమనాన్ని చేరిస్తే $\lforall[x][!A(x)]$కు
ఒక \tetoken{వ్యుత్పత్తిని}{!!a{derivation}} వస్తుంది. $c$ అనేది $\Gamma$లో గానీ $!A(x)$లో
గానీ కనిపించదు కనుక ఐగెన్ చరరాశి షరతు నెరవేరుతుంది.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}

\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{కిందిది $!A(t)$ నుంచి $\lexists[x][!A(x)]$కు
  ఒక \tetoken{వ్యుత్పత్తి}{!!a{derivation}}:
\begin{prooftree}
\AxiomC{$!A(t)$}
\RightLabel{\Intro{\lexists}}
\UnaryInfC{$\lexists[x][!A(x)]$}
\end{prooftree}}{}

\tagitem{prvAll}{కిందిది $\lforall[x][!A(x)]$ నుంచి $!A(t)$కు
  ఒక \tetoken{వ్యుత్పత్తి}{!!a{derivation}}:
\begin{prooftree}
\AxiomC{$\lforall[x][!A(x)]$}
\RightLabel{\Elim{\lforall}}
\UnaryInfC{$!A(t)$}
\end{prooftree}}{}
\end{tagenumerate}
\end{proof}
