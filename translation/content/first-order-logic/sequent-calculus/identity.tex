\olsection{\usetoken{S}{identity}తో \usetoken{P}{derivation}}

\tetoken{తాదాత్మ్య విధేయంతో}{!!{identity}} కూడిన \tetoken{వ్యుత్పత్తులకు}{!!^{derivation}s} అదనపు ప్రారంభ సీక్వెంట్లు,
నిగమన నియమాలు అవసరం.

\begin{defn}[$\eq$కు ప్రారంభ సీక్వెంట్లు]
$t$ ఒక సంవృత పదమైతే ${} \Sequent \eq[t][t]$ ఒక ప్రారంభ సీక్వెంట్.
\end{defn}

$\eq$కు నియమాలు కిందివి ($t_1$, $t_2$లు సంవృత పదాలు):

\begin{defish}
\Axiom$ \eq[t_1][t_2], \Gamma \fCenter \Delta, !A(t_1) $
\RightLabel{$\eq$}
\UnaryInf$\eq[t_1][t_2], \Gamma \fCenter \Delta, !A(t_2)$
\DisplayProof
\hfill
\Axiom$\eq[t_1][t_2], \Gamma \fCenter \Delta, !A(t_2) $
\RightLabel{$\eq$}
\UnaryInf$\eq[t_1][t_2], \Gamma  \fCenter \Delta, !A(t_1)$
\DisplayProof
\end{defish}

\begin{ex}
$s$, $t$లు సంవృత పదాలైతే $\eq[s][t], !A(s) \Proves !A(t)$:
\begin{prooftree}
\Axiom$ !A(s) \fCenter !A(s)$
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$\eq[s][t], !A(s)  \fCenter !A(s)$
\RightLabel{$\eq$}
\UnaryInf$\eq[s][t], !A(s)  \fCenter !A(t)$
\end{prooftree}
ఇది తాదాత్మ్య వస్తువుల ప్రతిస్థాపన సూత్రం లేదా లైబ్నిట్స్ నియమంగా
తెలిసి ఉండవచ్చు.

$\eq$ సౌష్ఠవమైనదీ, సంక్రామకమైనదీ అని $\Log{LK}$ నిరూపిస్తుంది:
\begin{prooftree}
\Axiom$ \fCenter \eq[t_1][t_1] $
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$ \eq[t_1][t_2] \fCenter \eq[t_1][t_1] $
\RightLabel{$\eq$}
\UnaryInf$ \eq[t_1][t_2] \fCenter \eq[t_2][t_1]$
\DisplayProof\qquad\bottomAlignProof
\Axiom$ \eq[t_1][t_2] \fCenter \eq[t_1][t_2] $
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$\eq[t_2][t_3], \eq[t_1][t_2]  \fCenter \eq[t_1][t_2] $
\RightLabel{$\eq$}
\UnaryInf$\eq[t_2][t_3], \eq[t_1][t_2]  \fCenter \eq[t_1][t_3]$
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$\eq[t_1][t_2], \eq[t_2][t_3]  \fCenter \eq[t_1][t_3]$
\end{prooftree}
ఎడమవైపున్న \tetoken{వ్యుత్పత్తిలో}{!!{derivation}} \tetoken{సూత్రం}{!!{formula}}~$\eq[x][t_1]$ మన $!A(x)$.
కుడివైపున $!A(x)$ను $\eq[t_1][x]$గా తీసుకుంటాం.
\end{ex}

\begin{prob}
కింది సీక్వెంట్లకు \tetoken{వ్యుత్పత్తులు}{!!{derivation}s} ఇవ్వండి:
\begin{enumerate}
\item $\Sequent \lforall[x][\lforall[y][((x = y \land !A(x)) \lif !A(y))]]$
\item $\lexists[x][!A(x)] \land \lforall[y][\lforall[z][((!A(y) \land
    !A(z)) \lif y = z)]] \Sequent
\lexists[x][(!A(x) \land \lforall[y][(!A(y) \lif y = x)])]$
\end{enumerate}
\end{prob}
