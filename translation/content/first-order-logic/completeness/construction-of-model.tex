\olsection{ఒక నమూనా నిర్మాణం}

\begin{explain}
\iftag{FOL}{ప్రస్తుతానికి~$\eq$ను పక్కన పెడదాం. అంటే
  $\eq$ను కలిగి లేని అవైరుధ్య \tetoken{వాక్యాల}{!!{sentence}s} సమితి~$\Gamma$
  సంతృప్తిపరచదగినదని మాత్రమే చూపాలి. మొదట~$\Gamma$ను
  అవైరుధ్యమైన, \tetoken{సంపూర్ణ}{!!{complete}}, సంతృప్తీకృత సమితి~$\Gamma^*$గా
  విస్తరిస్తాం. ఈ సందర్భంలో నమూనా~$\Struct{M(\Gamma^*)}$ను నిర్వచించడం
  సులభం. $\Lang{L'}$లోని సంవృత పదాల సమితిని వ్యక్తి క్షేత్రంగా తీసుకుంటాం.
  ప్రతి \tetoken{స్థిరసంకేతాన్ని}{!!{constant}} దానికే నిర్దేశిస్తాం. సాధారణంగా ప్రతి
  సంవృత పదం~$t$కు $\Value{t}{M(\Gamma^*)} = t$ అయ్యేలా చూస్తాం.
  ఒక పరమాణు \tetoken{వాక్యం}{!!{sentence}} $\Struct{M(\Gamma^*)}$లో సత్యమవడం, అది
  ~$\Gamma^*$లో ఉండటానికి తుల్యమయ్యేలా \tetoken{విధేయ సంకేతాలకు}{!!{predicate}s} విస్తరణలను
  నిర్దేశిస్తాం. ఇది~$\Gamma^*$లోని అన్ని పరమాణు \tetoken{వాక్యాలను}{!!{sentence}s}
  $\Struct{M(\Gamma^*)}$లో సత్యం చేస్తుందని స్పష్టం. మనం మొదలుపెట్టే
  $\Gamma^*$ అవైరుధ్యంగా, సంపూర్ణంగా, సంతృప్తీకృతంగా ఉంటే మిగిలినవి
  కూడా సత్యమవుతాయి.}{ఇప్పుడు ప్రతి $!A \in \Gamma$ను సత్యం చేసే
  \tetoken{ఒక సత్యమూల్య కేటాయింపును}{!!a{valuation}} నిర్వచించడానికి సిద్ధంగా ఉన్నాం. దీనికోసం మొదట
  లిండెన్‌బామ్ ఉపసిద్ధాంతాన్ని వర్తింపజేస్తాం: సంపూర్ణ అవైరుధ్య
  $\Gamma^* \supseteq \Gamma$ను పొందుతాం.
  ~$\Gamma^*$లోని \tetoken{ప్రతిజ్ఞావాక్య చరాలు}{!!{propositional variable}s}
  ~$\pAssign v(\Gamma^*)$ను నిర్ణయించేలా ఉంచుతాం.}
\end{explain}

\iftag{FOL}{%
\begin{defn}[పద నమూనా]
\ollabel{defn:termmodel}
భాష~$\Lang L$లోని \tetoken{వాక్యాల}{!!{sentence}s} \tetoken{ఒక సంపూర్ణ}{!!a{complete}}, అవైరుధ్య,
సంతృప్తీకృత సమితి~$\Gamma^*$ అనుకుందాం. $\Gamma^*$కు సంబంధించిన
\emph{పద నమూనా}~$\Struct M(\Gamma^*)$ కింది విధంగా నిర్వచించిన
\tetoken{నిర్మాణం}{!!{structure}}:
\begin{enumerate}
\item \tetoken{వ్యక్తి క్షేత్రం}{!!{domain}}~$\Domain{M(\Gamma^*)}$ అనేది~$\Lang L$లోని అన్ని
  సంవృత పదాల సమితి.
\item \tetoken{ఒక స్థిరసంకేతం}{!!a{constant}}~$c$కు ఇచ్చే అర్థవివరణ $c$యే:
  $\Assign{c}{M(\Gamma^*)} = c$.
\item సంవృత పదాలు $t_1$, \dots, $t_n$ను నివేశాలుగా తీసుకున్నప్పుడు
  సంవృత పదం $f(t_1, \dots, t_n)$ను విలువగా ఇచ్చే ప్రమేయాన్ని
  \tetoken{ప్రమేయ సంకేతం}{!!{function}}~$f$కు నిర్దేశిస్తాం:
\[
\Assign{f}{M(\Gamma^*)}(t_1, \dots, t_n) = f(t_1,\dots, t_n)
\]
\item $R$ ఒక $n$-స్థాన \tetoken{విధేయ సంకేతం}{!!{predicate}} అయితే,
  \[
  \tuple{t_1, \dots,
  t_n} \in \Assign{R}{M(\Gamma^*)} \text{ అప్పుడూ అప్పుడే } \Atom{R}{t_1, \dots,
    t_n} \in \Gamma^*.
  \]
\end{enumerate}
\end{defn}
}{%
\begin{defn}
\ollabel{defn:termmodel}
$\Gamma^*$ ఒక సంపూర్ణ అవైరుధ్య \tetoken{సూత్రాల}{!!{formula}s} సమితి అనుకుందాం. అప్పుడు
\[
\pAssign v(\Gamma^*)(p) = \begin{cases}
  \True & \text{ఒకవేళ $p \in \Gamma^*$}\\
  \False & \text{ఒకవేళ $p \notin \Gamma^*$}
  \end{cases}
\]
గా ఉంచుతాం.
\end{defn}
}

\iftag{FOL}{%
నిజంగానే $\Value{t}{M(\Gamma^*)} = t$ అని ఇప్పుడు చూద్దాం.

\begin{lem}
\ollabel{lem:val-in-termmodel} $\Struct M(\Gamma^*)$ అనేది
\olref{defn:termmodel}లోని పద నమూనా అయితే, ప్రతి సంవృత పదానికీ
$\Value{t}{M(\Gamma^*)} = t$.
\end{lem}
\begin{proof}
$t$పై ఆగమనం ద్వారా నిరూపిస్తాం. ఆగమనపు తొలి మెట్టులో $t$ ఒక
\tetoken{స్థిరసంకేతం}{!!a{constant}}. కావలసినది పద నమూనా నిర్వచనం నుంచి నేరుగా వస్తుంది. ఆగమన
అడుగులో $t_1, \ldots, t_n$లు $\Value{t_i}{M(\Gamma^*)} = t_i$ అయ్యే
సంవృత పదాలని, $f$ ఒక $n$-స్థాన \tetoken{ప్రమేయ సంకేతం}{!!{function}} అని అనుకుందాం. అప్పుడు
\begin{align*}
\Value{f(t_1,\ldots,t_n)}{M(\Gamma^*)} &= \Assign{f}{M(\Gamma^*)}(\Value{t_1}
 {M(\Gamma^*)},
\ldots, \Value{t_n}{M(\Gamma^*)}) \\
&= \Assign{f}{M(\Gamma^*)}(t_1, \dots, t_n) \\
&= f(t_1,\dots, t_n),
\end{align*}
కనుక ఆగమనం ద్వారా ప్రతి సంవృత పదం~$t$కు ఇది వర్తిస్తుంది.
\end{proof}
}{}

\begin{explain}
  \iftag{prvEx}{ఏదైనా \tetoken{ఒక నిర్మాణం}{!!^a{structure}}~$\Struct{M}$ అస్తిత్వంగా
    పరిమాణీకృత \tetoken{వాక్యం}{!!{sentence}}~$\lexists[x][!A(x)]$ను సత్యం చేసినా, అది
    సత్యం చేసే దృష్టాంతం~$!A(t)$ ఏదీ లేకపోవచ్చు.}{}

    పరిమాణీకృత \tetoken{వాక్యం}{!!{sentence}}~$\lforall[x][!A(x)]$కు సంబంధించిన అన్ని
    దృష్టాంతాలైన~$!A(t)$ను సత్యం చేసినా, $\lforall[x][!A(x)]$ను సత్యం
    చేయకపోవచ్చు.}{} సాధారణంగా~$\Domain{M}$లో ఏ సంవృత పదానికీ విలువ కాని
  \tetoken{మూలకాలు}{!!{element}} ఉండవచ్చు. ఇదే దీనికి కారణం ($\Struct{M}$ ఆవరితమై
  ఉండకపోవచ్చు). సంతృప్తి సంబంధాన్ని చరరాశుల విలువనిర్ణయాల ద్వారా
  నిర్వచించడానికి ఇదే కారణం. అయితే మన పద నమూనా~$\Struct{M(\Gamma^*)}$కు
  ఇది అవసరం కాదు---ఎందుకంటే అది ఆవరితమైనది. ఇదే తరువాతి ఫలితపు విషయం.
\end{explain}

\begin{prop}
\ollabel{prop:quant-termmodel}
$\Struct M(\Gamma^*)$ అనేది \olref{defn:termmodel}లోని పద నమూనా
అనుకుందాం.
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$\Sat{M(\Gamma^*)}{\lexists[x][!A(x)]}$ అయినప్పుడు,
  అప్పుడు మాత్రమే కనీసం ఒక సంవృత పదం~$t$కు
  $\Sat{M(\Gamma^*)}{!A(t)}$.}{}
\tagitem{prvAll}{$\Sat{M(\Gamma^*)}{\lforall[x][!A(x)]}$ అయినప్పుడు,
  అప్పుడు మాత్రమే ప్రతి సంవృత పదం~$t$కూ
  $\Sat{M(\Gamma^*)}{!A(t)}$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{%
  \iftag{probEx}{అభ్యాసం.}{\olref[syn][ass]{prop:sat-quant} ప్రకారం,
    కనీసం ఒక చరరాశి విలువనిర్ణయం~$s$కు
    $\Sat{M(\Gamma^*)}{!A(x)}[s]$ అయినప్పుడు, అప్పుడు మాత్రమే
    $\Sat{M(\Gamma^*)}{\lexists[x][!A(x)]}$. $\Domain{M(\Gamma^*)}$
    అనేది~$\Lang{L}$లోని సంవృత పదాల సమితి కనుక, $s(x) = t$, ఇంకా
    $\Sat{M(\Gamma^*)}{!A(x)}[s]$ అయ్యే కనీసం ఒక సంవృత పదం~$t$
    ఉన్నప్పుడు, అప్పుడు మాత్రమే ఇది జరుగుతుంది.
    \olref[fol][syn][ext]{prop:ext-formulas} ప్రకారం, $s(x) = t$
    అయినప్పుడు $\Sat{M(\Gamma^*)}{!A(x)}[s]$ అయినప్పుడు, అప్పుడు
    మాత్రమే $\Sat{M(\Gamma^*)}{!A(t)}[s]$. $!A(t)$ ఒక వాక్యం కనుక,
    \olref[fol][syn][ass]{prop:sentence-sat-true} ప్రకారం
    $\Sat{M(\Gamma^*)}{!A(t)}[s]$ అయినప్పుడు, అప్పుడు మాత్రమే
    $\Sat{M(\Gamma^*)}{!A(t)}$.}}{}
\tagitem{prvAll}{%
  \iftag{probAll}{అభ్యాసం.}{\olref[syn][ass]{prop:sat-quant} ప్రకారం,
    ప్రతి చరరాశి విలువనిర్ణయం~$s$కు $\Sat{M(\Gamma^*)}{!A(x)}[s]$
    అయినప్పుడు, అప్పుడు మాత్రమే
    $\Sat{M(\Gamma^*)}{\lforall[x][!A(x)]}$. $\Domain{M(\Gamma^*)}$
    అనేది~$\Lang{L}$లోని సంవృత పదాల సమితి అని గుర్తుంచుకోండి.
    కాబట్టి ప్రతి సంవృత పదం~$t$కు $s(x) = t$ అయ్యే చరరాశి విలువనిర్ణయం
    ఉంటుంది. ఏ చరరాశి విలువనిర్ణయానికైనా $s(x)$ ఏదో ఒక సంవృత పదం~$t$.

\begin{lem}[సత్య ఉపసిద్ధాంతం]
\ollabel{lem:truth} \iftag{FOL}{$!A$లో~$\eq$ లేదని అనుకుందాం. అప్పుడు}{}
$\iftag{FOL}{\Sat{M(\Gamma^*)}}{\pSat{v(\Gamma^*)}}{!A}$ అయినప్పుడు,
అప్పుడు మాత్రమే $!A \in \Gamma^*$.
\end{lem}

\begin{proof}
రెండు దిశలనూ ఒకేసారి, $!A$పై ఆగమనంతో నిరూపిస్తాం.
\begin{enumerate}
\tagitem{prvFalse}{\indcase{!A}{\lfalse}
  {$\iftag{FOL}{\Sat/{M(\Gamma^*)}{\lfalse}}{\pSat/{v(\Gamma^*)}{\lfalse}}$
    అని సంతృప్తి నిర్వచనం చెబుతుంది. మరోవైపు, $\Gamma^*$ అవైరుధ్యమైనది
    కనుక $\lfalse \notin \Gamma^*$.}}{}

\tagitem{prvTrue}{\indcase{!A}{\ltrue}
  {$\iftag{FOL}{\Sat{M(\Gamma^*)}}{\pSat{v(\Gamma^*)}}{\ltrue}$
    అని సంతృప్తి నిర్వచనం చెబుతుంది. మరోవైపు, $\Gamma^*$ అవైరుధ్యమైనది,
    \tetoken{సంపూర్ణమైనది}{!!{complete}}, ఇంకా $\Gamma^* \Proves \ltrue$ కనుక $\ltrue \in
    \Gamma^*$.}}{}

\tagitem{FOL}{\indcase{!A}{R(t_1, \dots, t_n)}
  {$\Sat{M(\Gamma^*)}{\Atom{R}{t_1, \dots, t_n}}$ అయినప్పుడు, అప్పుడు
    మాత్రమే $\tuple{t_1, \dots, t_n} \in \Assign{R}{M(\Gamma^*)}$
    (సంతృప్తి నిర్వచనం ప్రకారం). అలాగే అది జరిగితే, ఆ సందర్భంలో మాత్రమే
    $R(t_1, \dots, t_n) \in \Gamma^*$ ($\Struct M(\Gamma^*)$
    నిర్మాణం ప్రకారం).}}{\indcase{!A}{p}{$\pSat{v(\Gamma^*)}{p}$
    అయినప్పుడు, అప్పుడు మాత్రమే $\pAssign v(\Gamma^*)(p) = \True$
    (సంతృప్తి నిర్వచనం ప్రకారం). అలాగే అది జరిగితే, ఆ సందర్భంలో మాత్రమే
    $p \in \Gamma^*$ ($\pAssign v(\Gamma^*)$ నిర్మాణం ప్రకారం).}}

\tagitem{prvNot}{%
  \indcase{!A}{\lnot !B}
    {$\iftag{FOL}{\Sat{M(\Gamma^*)}}{\pSat{v(\Gamma^*)}}{\indfrm}$
    అయినప్పుడు, అప్పుడు మాత్రమే
    $\iftag{FOL}{\Sat/{M(\Gamma^*)}{!B}}{\pSat/{v(\Gamma^*)}{!B}}$
    (సంతృప్తి నిర్వచనం ప్రకారం). ఆగమన పరికల్పన ప్రకారం,
    $\iftag{FOL}{\Sat/{M(\Gamma^*)}{!B}}{\pSat/{v(\Gamma^*)}{!B}}$
    అయినప్పుడు, అప్పుడు మాత్రమే $!B \notin \Gamma^*$. $\Gamma^*$
    అవైరుధ్యమైనది, \tetoken{సంపూర్ణమైనది}{!!{complete}} కనుక $!B \notin \Gamma^*$ అయినప్పుడు,
    అప్పుడు మాత్రమే $\lnot !B \in \Gamma^*$.}}{}

\end{proof}
