\olsection{పరిమాణీకరణలతో \usetoken{P}{derivation}}

\begin{ex}
$(\lforall[x][!A(x)] \land \lforall[y][!B(y)]) \lif
\lforall[x][(!A(x) \land !B(x))]$కు ఒక \tetoken{వ్యుత్పత్తిని}{!!a{derivation}} ఇద్దాం.

ముందుగా కింది దృష్టాంతాలను గమనించండి.
\begin{align*}
  (\lforall[x][!A(x)] \land \lforall[y][!B(y)]) & \lif \lforall[x][!A(x)]\\
  \intertext{అనేది \olref[prp]{ax:land1} దృష్టాంతం. అలాగే}
  \lforall[x][!A(x)] & \lif !A(a) \\
  \intertext{అనేది \olref[qua]{ax:q1} దృష్టాంతం. కాబట్టి \olref[pro]{prop:chain} ప్రకారం}
  (\lforall[x][!A(x)] \land \lforall[y][!B(y)]) & \lif !A(a) \\
  \intertext{\tetoken{వ్యుత్పాదించదగినదని}{!!{derivable}} తెలుస్తుంది. అదే విధంగా,}
  (\lforall[x][!A(x)] \land \lforall[y][!B(y)]) & \lif \lforall[y][!B(y)] \qquad\text{మరియు}\\
    \lforall[y][!B(y)] & \lif !B(a)\\
    \intertext{అనేవి వరుసగా \olref[prp]{ax:land2}, \olref[qua]{ax:q1} దృష్టాంతాలు కనుక,}
    (\lforall[x][!A(x)] \land \lforall[y][!B(y)]) & \lif !B(a)\\
    \intertext{అనేది \olref[pro]{prop:chain} వల్ల వ్యుత్పాదించదగినది. \olref[prp]{ax:land3}కు తగిన ఒక దృష్టాంతాన్ని తీసుకుని,~\MPని రెండుసార్లు ఉపయోగిస్తే}
    (\lforall[x][!A(x)] \land \lforall[y][!B(y)]) & \lif (!A(a) \land !B(a))\\
    \intertext{వ్యుత్పాదించదగినదని తెలుస్తుంది. ఇప్పుడు \QR{}ను వర్తింపజేస్తే}
(\lforall[x][!A(x)] \land \lforall[y][!B(y)]) & \lif \lforall[x][(!A(x) \land !B(x))].
\end{align*}
\end{ex}
