\olsection{\usetoken{S}{identity}తో \usetoken{P}{derivation}}

\tetoken{వ్యుత్పత్తులలో}{!!{derivation}s} $\eq$ను చేర్చడానికి కొత్త స్వీకృత
రూపాలను చేర్చితే చాలు. \tetoken{వ్యుత్పత్తి}{!!{derivation}}, $\Proves$ల నిర్వచనం
అదే ఉంటుంది. కొత్త స్వీకృతాలను కూడా అనుమతిస్తాం.

\begin{defn}[\tetoken{తాదాత్మ్య విధేయం}{!!{identity}} స్వీకృతాలు]
\begin{align}
\ollabel{ax:id1} & \eq[t][t], \\
\ollabel{ax:id2} & \eq[t_1][t_2] \lif (!B(t_1) \lif
  !B(t_2)),
\end{align}
ఇక్కడ $t$, $t_1$, $t_2$ ఏ సంవృత పదాలైనా కావచ్చు.
\end{defn}

\begin{prop}\ollabel{prop:sound}
  \olref{ax:id1}, \olref{ax:id2} స్వీకృతాలు చెల్లుబాటు అవుతాయి.
\end{prop}

\begin{proof}
  అభ్యాసం.
\end{proof}

\begin{prob}
\olref[fol][axd][ide]{prop:sound}ను నిరూపించండి.
\end{prob}

\begin{prop}\ollabel{prop:iden1}
 ఏ సంవృత పదం~$t$, ఏ సమితి~$\Gamma$కైనా $\Gamma \Proves \eq[t][t]$.
\end{prop}

\begin{prop}\ollabel{prop:iden2} $t_1$, $t_2$ సంవృత పదాలు అనుకుందాం.
  $\Gamma \Proves !A(t_1)$, $\Gamma \Proves \eq[t_1][t_2]$ అయితే,
  $\Gamma \Proves !A(t_2)$.
\end{prop}

\begin{proof}
\tetoken{సూత్రం}{!!{formula}}
\[
(\eq[t_1][t_2] \lif (!A(t_1) \lif !A(t_2)))
\]
అనేది~\olref{ax:id2}కు ఒక దృష్టాంతం. \MPని రెండుసార్లు వర్తింపజేస్తే
ఈ నిష్కర్ష వస్తుంది.
\end{proof}
