\olsection{నిరూపణ-సిద్ధాంత భావనలు}

\begin{explain}
మనం అనేక ముఖ్యమైన అర్థపర భావనలను
(\iftag{FOL}{చెల్లుబాటు}{సర్వసత్యత}, అనుగమనం, సంతృప్తత)
నిర్వచించినట్లే, ఇప్పుడు వాటికి సరిపోలే \emph{నిరూపణ-సిద్ధాంత
భావనలను} నిర్వచిస్తాం. వీటిని \tetoken{నిర్మాణాలలో}{!!{structure}s} \tetoken{వాక్యాలు}{!!{sentence}s}
సంతృప్తి చెందడాన్ని బట్టి నిర్వచించం. కొన్ని సూత్రాల
\tetoken{వ్యుత్పాద్యతను}{!!{derivability}} లేదా \tetoken{అవ్యుత్పాద్యతను}{!!{nonderivability}} బట్టి నిర్వచిస్తాం.
ఈ భావనలు ఏకీభవిస్తాయని కనుగొనడం ఒక ముఖ్యమైన ఆవిష్కరణ. ఆ ఏకీభావాన్నే \emph{నిర్దుష్టత}, \emph{సంపూర్ణత}
సిద్ధాంతాలు చెబుతాయి.
\end{explain}

\begin{defn}[\tetoken{వ్యుత్పాద్యత}{!!^{derivability}}]
$\Gamma$ నుంచి $!A$తో ముగిసే ఒక \tetoken{వ్యుత్పత్తి}{!!a{derivation}} ఉంటే,
\tetoken{ఒక సూత్రం}{!!^a{formula}}~$!A$ను $\Gamma$ నుంచి \emph{\tetoken{వ్యుత్పాదించదగినది}{!!{derivable}}} అంటాం.
దీన్ని $\Gamma \Proves !A$ అని రాస్తాం.
\end{defn}

\begin{defn}[సిద్ధాంతాలు]
ఖాళీ సమితి నుంచి $!A$కు ఒక \tetoken{వ్యుత్పత్తి}{!!a{derivation}} ఉంటే,
\tetoken{ఒక సూత్రం}{!!^a{formula}}~$!A$ ఒక \emph{సిద్ధాంతం}. $!A$ సిద్ధాంతమైతే
$\Proves !A$ అని, కాకపోతే $\Proves/ !A$ అని రాస్తాం.
\end{defn}

\begin{defn}[అవైరుధ్యం]
\tetoken{సూత్రాల}{!!{formula}s} సమితి~$\Gamma$ $\Gamma\Proves/ \lfalse$ అయితే,
ఆ సందర్భంలో మాత్రమే \emph{అవైరుధ్యమైనది}. లేకపోతే అది \emph{వైరుధ్యభరితమైనది}.
\end{defn}

\begin{prop}[స్వావర్తనత్వం]
\ollabel{prop:reflexivity}
$!A \in \Gamma$ అయితే, $\Gamma \Proves !A$.
\end{prop}

\begin{proof}
  \tetoken{సూత్రం}{!!{formula}}~$!A$ ఒక్కటే~$\Gamma$ నుంచి~$!A$కు ఒక
  \tetoken{వ్యుత్పత్తి}{!!a{derivation}}.
\end{proof}

\begin{prop}[ఏకదిశత]
\ollabel{prop:monotonicity}
$\Gamma \subseteq \Delta$, $\Gamma \Proves !A$ అయితే,
$\Delta \Proves !A$.
\end{prop}

\begin{proof}
$\Gamma$ నుంచి $!A$కు ఉన్న ఏ \tetoken{వ్యుత్పత్తి}{!!{derivation}} అయినా~$\Delta$ నుంచి
$!A$కు కూడా ఒక \tetoken{వ్యుత్పత్తి}{!!a{derivation}}.
\end{proof}

\begin{prop}[సంక్రామకత్వం]
\ollabel{prop:transitivity}
$\Gamma \Proves !A$, $\{!A\} \cup \Delta \Proves !B$ అయితే,
$\Gamma \cup \Delta \Proves !B$.
\end{prop}

\begin{proof}
  $\{!A\} \cup \Delta \Proves !B$ అని అనుకుందాం. అప్పుడు
  $\{!A\} \cup \Delta$ నుంచి~$!B_1$, \dots, $!B_l = !B$ అనే ఒక
  \tetoken{వ్యుత్పత్తి}{!!a{derivation}} ఉంది. ఆ \tetoken{వ్యుత్పత్తిలోని}{!!{derivation}} కొన్ని దశలు, ముందరి
  పంక్తి~$!B_i = !A$ను సూచించే నియమం వల్ల సరైనవిగా ఉంటాయి.
  పరికల్పన ప్రకారం,~$\Gamma$ నుంచి~$!A$కు ఒక \tetoken{వ్యుత్పత్తి}{!!a{derivation}} ఉంది.
  అందులో ప్రతి $!A_i$ ఒక స్వీకృతమో,~$\Gamma$లోని ఒక
  \tetoken{మూలకం}{!!a{element}} అయి ఉంటుంది లేదా నిగమన నియమం ద్వారా వస్తుంది.
  ఆ \tetoken{వ్యుత్పత్తిని}{!!a{derivation}}~$!A_1$, \dots, $!A_k = !A$ అని రాద్దాం. ఇప్పుడు ఈ క్రమాన్ని
  పరిశీలించండి:
  \[
  !A_1, \dots, !A_k = !A, !B_1, \dots, !B_l = !B.
  \]
  ఇది $\Gamma \cup \Delta$ నుంచి~$!B$కు సరైన \tetoken{వ్యుత్పత్తి}{!!{derivation}}. ఎందుకంటే
  ప్రతి $!B_i = !A$ ఇప్పుడు~$!A_k = !A$ను సమర్థించే అదే ఆధారంతో
  సమర్థించబడుతుంది.
\end{proof}

దీనర్థం, ముఖ్యంగా, $\Gamma \Proves !A$, $!A \Proves !B$ అయితే
$\Gamma \Proves !B$. అలాగే, $!A_1, \dots, !A_n \Proves !B$ అయి,
ప్రతి~$i$కు $\Gamma \Proves !A_i$ అయితే, $\Gamma \Proves !B$ అని
కూడా అనుగమిస్తుంది.

\begin{prop}
\ollabel{prop:incons}
ప్రతి~$!A$కు $\Gamma \Proves !A$ అయితే, అప్పుడు మాత్రమే $\Gamma$
వైరుధ్యభరితమైనది.
\end{prop}

\begin{proof}
అభ్యాసం.
\end{proof}

\begin{prop}[సంహతత్వం]
\ollabel{prop:proves-compact}
  \begin{enumerate}
  \item $\Gamma \Proves !A$ అయితే, $\Gamma_0 \Proves !A$ అయ్యే
    పరిమిత ఉపసమితి $\Gamma_0 \subseteq \Gamma$ ఒకటి ఉంటుంది.
  \item~$\Gamma$లోని ప్రతి పరిమిత ఉపసమితీ అవైరుధ్యమైనదైతే,
    $\Gamma$ అవైరుధ్యమైనది.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item $\Gamma \Proves !A$ అయితే, $!A \ident !A_n$ అయ్యేలా,
      అలాగే ప్రతి $!A_i$ ఒక తార్కిక స్వీకృతమో,~$\Gamma$లోని ఒక
      \tetoken{మూలకం}{!!a{element}} అయి ఉండేదో, ముందున్న \tetoken{సూత్రాల}{!!{formula}s} నుంచి అనుమతించిన నిగమన నియమం
      ద్వారా వచ్చేదో అయ్యేలా $!A_1$, \dots,~$!A_n$ అనే పరిమిత
      \tetoken{సూత్రాల}{!!{formula}s} క్రమం ఉంటుంది. వీటిలో~$\Gamma$లో ఉన్న $!A_i$ల
      సమితినే $\Gamma_0$గా తీసుకోండి. అప్పుడు ఆ \tetoken{వ్యుత్పత్తి}{!!{derivation}},
      $\Gamma_0$ నుంచి కూడా ఒక \tetoken{వ్యుత్పత్తి}{!!a{derivation}}. అందువల్ల
      $\Gamma_0 \Proves !A$.
    \item $!A \ident \lfalse$ అనే ప్రత్యేక సందర్భంలో ఇది~(1)కు
      విపర్యయ ప్రతిజ్ఞ.
\end{enumerate}
\end{proof}
