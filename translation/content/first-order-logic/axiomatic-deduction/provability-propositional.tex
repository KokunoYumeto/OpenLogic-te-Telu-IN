\olsection{\usetoken{S}{derivability}, ప్రతిజ్ఞావాక్య సంయోజకాలు}

\begin{explain}
  $!A \land !B \Proves !A$, $!A, !A \lif !B \Proves !B$ (మోడస్
  పోనెన్స్) వంటి ప్రతిజ్ఞావాక్య సంయోజకాలకు సంబంధించిన కొన్ని ప్రాథమిక
  ఫలితాలను నిరూపించడానికి స్వీకృతాధారిత నిగమనంలోని
  \tetoken{వ్యుత్పాద్యత}{!!{derivability}} సంబంధం~$\Proves$ తగినంత బలమైనదని నిరూపిస్తాం.
  సంపూర్ణతా సిద్ధాంతపు నిరూపణకు ఈ ఫలితాలు అవసరం.
\end{explain}

\begin{prop}\ollabel{prop:provability-land}
  \begin{enumerate}
  \item \ollabel{prop:provability-land-left} $!A \land !B \Proves
    !A$, $!A \land !B \Proves !B$ రెండూ నెరవేరుతాయి.
  \item \ollabel{prop:provability-land-right} $!A, !B \Proves !A \land !B$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item \olref[prp]{ax:land1}, \olref[prp]{ax:land2}ల నుంచి మోడస్
      పోనెన్స్ ద్వారా.
  \item \olref[prp]{ax:land3} నుంచి మోడస్ పోనెన్స్‌ను రెండుసార్లు
    వర్తింపజేయడం ద్వారా.
  \end{enumerate}
\end{proof}

\begin{prop}\ollabel{prop:provability-lor}
  \begin{enumerate}
  \item $!A \lor !B, \lnot !A, \lnot !B$ వైరుధ్యభరితమైనది.
  \item $!A \Proves !A \lor !B$, $!B \Proves !A \lor !B$ రెండూ నెరవేరుతాయి.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item \olref[prp]{ax:lnot2} నుంచి $\Proves \lnot !A \lif (!A
    \lif \lfalse)$, $\Proves \lnot !B \lif (!B \lif \lfalse)$
    లభిస్తాయి.
    కాబట్టి నిగమన సిద్ధాంతం వల్ల $\{\lnot !A\} \Proves
    !A \lif \lfalse$, $\{\lnot !B\} \Proves !B \lif \lfalse$.
    \olref[prp]{ax:lor3} నుంచి $\{\lnot !A, \lnot !B\} \Proves
    (!A \lor !B) \lif \lfalse$ లభిస్తుంది. నిగమన సిద్ధాంతం వల్ల,
    $\{!A \lor !B, \lnot !A, \lnot !B\} \Proves \lfalse$.
  \item \olref[prp]{ax:lor1}, \olref[prp]{ax:lor2}ల నుంచి మోడస్
    పోనెన్స్ ద్వారా.
  \end{enumerate}
\end{proof}

\begin{prop}\ollabel{prop:provability-lif}
  \begin{enumerate}
  \item \ollabel{prop:provability-lif-left} $!A, !A \lif !B \Proves !B$.
  \item \ollabel{prop:provability-lif-right}
    $\lnot !A \Proves !A \lif !B$, $!B \Proves !A \lif !B$ రెండూ నెరవేరుతాయి.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item మనం ఇలా \tetoken{వ్యుత్పాదించవచ్చు}{!!{derive}}:
    \begin{derivation}
      1. & $!A$ & \Hyp\\
      2. & $!A \lif !B$ & \Hyp\\
      3. & $!B$ & 1, 2, \MP
    \end{derivation}

  \item వరుసగా \olref[prp]{ax:lnot2}, \olref[prp]{ax:lif1}, నిగమన
    సిద్ధాంతం ద్వారా.
  \end{enumerate}
\end{proof}
