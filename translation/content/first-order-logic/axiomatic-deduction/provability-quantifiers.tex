\olsection{\usetoken{S}{derivability}, పరిమాణీకరణలు}

\begin{explain}
  ఈ విభాగంలో నిరూపించే~$\Proves$ సంబంధానికి సంబంధించిన ఫలితాలు స్వీకృతాధారిత
  నిగమనం నుంచి రావాలి. ఇది కూడా సంపూర్ణతా సిద్ధాంతానికి అవసరం.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} $c$ అనేది $\Gamma$లో గానీ
$!A(x)$లో గానీ కనిపించని ఒక \tetoken{స్థిరసంకేతం}{!!a{constant}} అయి, $\Gamma \Proves
!A(c)$ అయితే, $\Gamma \Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
నిగమన సిద్ధాంతం వల్ల $\Gamma \Proves \ltrue \lif !A(c)$. $c$ అనేది
$\Gamma$లో గానీ~$\top$లో గానీ కనిపించదు కనుక, $\Gamma \Proves
\ltrue \lif \lforall[x][!A(x)]$. $\Gamma \Proves \ltrue$ అనేది
\texttt{ax:ltrue} స్వీకృత రూపం. కాబట్టి మోడస్ పోనెన్స్ వల్ల $\Gamma
\Proves \lforall[x][!A(x)]$.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers} $t$ ఒక సంవృత పదం అనుకుందాం.
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}

\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{\olref[qua]{ax:q2}, నిగమన సిద్ధాంతం ద్వారా.}{}
\tagitem{prvAll}{\olref[qua]{ax:q1}, నిగమన సిద్ధాంతం ద్వారా.}{}
\end{tagenumerate}
\end{proof}
