\olsection{ఒక లోకంలో సత్యం}

ఒక మోడల్ నమూనాలో ఏ లోకంలో ఏ మోడల్
\tetoken{సూత్రాలు}{!!{formula}s} సత్యమో ఆ నమూనా నిర్ణయిస్తుంది.
“నమూనా $\mModel{M}$లో \tetoken{సూత్రం}{!!{formula}}~$!A$ అనేది
లోకం~$w$లో సత్యం” అనే సంబంధం సంబంధాత్మక
అర్థవిచారానికి మౌలికం. దాన్ని ఆగమన పద్ధతిలో నిర్వచిస్తాం.
మోడల్ కాని సంచాలకాల విషయంలో, ఈ నిర్వచనం
సాధారణ సత్య పట్టికలు ఇచ్చే షరతులతో సరిపోతుంది.

\begin{defn}\ollabel{defn:mmodels}
  నమూనా $\mModel M$లో \emph{\tetoken{సూత్రం}{!!a{formula}}~$!A$
  లోకం $w$లో సత్యం} అనే విషయాన్ని $\mSat{M}{!A}[w]$ అని
  రాస్తాం. దాన్ని ఆగమన పద్ధతిలో కింది విధంగా నిర్వచిస్తాం.
  \begin{enumerate}
  \tagitem{prvFalse}{\indcase{!A}{\lfalse}{ఎప్పుడూ
    $\mSat{M}{\lfalse}[w]$ కాదు}.}{}
  \tagitem{prvTrue}{\indcase{!A}{\ltrue}{ఎప్పుడూ
    $\mSat{M}{\ltrue}[w]$ నిలుస్తుంది}.}{}
  \item $\mSat{M}{p}[w]$ వర్తించేది $w \in V(p)$ అయినప్పుడు, అప్పుడే మాత్రమే.
  \tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$
    వర్తించేది
    $\mSat/{M}{!B}[w]$ అయినప్పుడు, అప్పుడే మాత్రమే}.}{}
  \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\mSat{M}{\indfrm}[w]$
    వర్తించేది
    $\mSat{M}{!B}[w]$, $\mSat{M}{!C}[w]$ రెండూ వర్తించినప్పుడు, అప్పుడే మాత్రమే}.}{}
  \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\mSat{M}{\indfrm}[w]$
    వర్తించేది
    $\mSat{M}{!B}[w]$, $\mSat{M}{!C}[w]$లలో కనీసం ఒకటి వర్తించినప్పుడూ, వర్తించినప్పుడు మాత్రమే} (రెండూ కావచ్చు).}{}
  \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\mSat{M}{\indfrm}[w]$
    వర్తించేది
    $\mSat/{M}{!B}[w]$, $\mSat{M}{!C}[w]$లలో కనీసం ఒకటి వర్తించినప్పుడూ, వర్తించినప్పుడు మాత్రమే}.}{}
  \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\mSat{M}{\indfrm}[w]$
    వర్తించడానికి అవసరమైన, సరిపడే షరతు
    $\mSat{M}{!B}[w]$, $\mSat{M}{!C}[w]$
    రెండూ వర్తించడమో లేదా
    $\mSat{M}{!B}[w]$, $\mSat{M}{!C}[w]$
    రెండూ వర్తించకపోవడమో}.}{}
  \item\ollabel{defn:sub:mmodels-box}
    \indcase{!A}{\Box !B}{$\mSat{M}{\indfrm}[w]$
    వర్తించడానికి అవసరమైన, సరిపడే షరతు $Rww'$
    ఉన్న ప్రతి $w' \in W$కూ
    $\mSat{M}{!B}[w']$ వర్తించడం.}
  \item\ollabel{defn:sub:mmodels-diamond}
    \indcase{!A}{\Diamond !B}{$\mSat{M}{\indfrm}[w]$
    వర్తించడానికి అవసరమైన, సరిపడే షరతు $Rww'$
    ఉన్న కనీసం ఒక $w' \in W$కు
    $\mSat{M}{!B}[w']$ వర్తించడం.}
  \end{enumerate}
\end{defn}

\olref{defn:sub:mmodels-box} ప్రకారం
$Rww'$ అయ్యే $w'$ ఏదీ లేకపోతే,
\tetoken{సూత్రం}{!!a{formula}}~$\Box
!B$ అనేది $w$లో సత్యం. అప్పుడు $\Box !B$ను
$w$లో \emph{శూన్యసందర్భ సత్యం} అంటాం.
$!B$ అక్కడ అసత్యమైనా $\Box !B$ అనేది $w$లో
సత్యం కావచ్చు. అలాగే $!B$ అనేది $w$లో సత్యమైనంత
మాత్రాన $\Diamond !B$ కూడా $w$లో సత్యమని తేలదు.
అయితే $Rww$ ఉన్నప్పుడు, ఉదాహరణకు $R$ స్వావర్తన
సంబంధమైతే, ఆ అనుగమనం వర్తిస్తుంది. $Rww'$ అయ్యే $w'$ ఏదీ లేకపోతే,
ఏ $!A$కైనా $\mSat/{M}{\Diamond !A}[w]$.

\begin{prob}
  \olref[nml][syn][rel]{fig:simple}లోని
  నమూనాను పరిశీలించండి. కింది
  వాటిలో ఏవి వర్తిస్తాయి?
    \begin{enumerate}
    \item $\mSat{M}{q}[w_1]$;
    \item $\mSat{M}{\lnot q}[w_3]$;
    \item $\mSat{M}{p \lor q}[w_1]$;
    \item $\mSat{M}{\Box (p \lor q)}[w_1]$;
    \item $\mSat{M}{\Box q}[w_3]$;
    \item $\mSat{M}{\Box \bot}[w_3]$;
    \item $\mSat{M}{\Diamond q}[w_1]$;
    \item $\mSat{M}{\Box q}[w_1]$;
    \item $\mSat{M}{\lnot \Box\Box \lnot q}[w_1]$.
    \end{enumerate}
\end{prob}

% Prove duality of \Box and \Diamond
\begin{tagblock}{prvDiamond}
\begin{prop}\ollabel{prop:dual}
  \begin{enumerate}
  \item $\mSat{M}{\Box !A}[w]$ వర్తించడానికి అవసరమైన, సరిపడే షరతు
    $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ వర్తించడం.
  \item $\mSat{M}{\Diamond !A}[w]$ వర్తించడానికి అవసరమైన, సరిపడే షరతు
    $\mSat{M}{\lnot\Box\lnot !A}[w]$ వర్తించడం.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item $\mSat{M}{\lnot\Diamond\lnot !A}[w]$
      అనేదానికి సమానమైన షరతు
      $\mSat/{M}{\Diamond\lnot !A}[w]$.
      ఇది $\mSat{M}{}[w]$ నిర్వచనం వల్ల.
      $\mSat{M}{\Diamond\lnot !A}[w]$
      వర్తించడానికి అవసరమైన, సరిపడే షరతు
      $Rww'$ ఉన్న ఏదో $w'$లో
      $\mSat{M}{\lnot !A}[w']$ వర్తించడం.
      కాబట్టి
      $\mSat/{M}{\Diamond\lnot !A}[w]$
      వర్తించడానికి అవసరమైన, సరిపడే షరతు
      $Rww'$ ఉన్న ప్రతి $w'$లో
      $\mSat/{M}{\lnot !A}[w']$ వర్తించడం.
      అలాగే $\mSat/{M}{\lnot !A}[w']$
      అనేది
      $\mSat{M}{!A}[w']$కు సమానం.
      వీటిని కలిపితే
      $\mSat{M}{\lnot\Diamond\lnot !A}[w]$
      వర్తించడానికి అవసరమైన, సరిపడే షరతు
      $Rww'$ ఉన్న ప్రతి $w'$లో
      $\mSat{M}{!A}[w']$ వర్తించడం.
      మళ్లీ $\mSat{M}{}[w]$
      నిర్వచనం ప్రకారం అది
      $\mSat{M}{\Box !A}[w]$కు
      సమానం.
    \item \iftag{probDiamond}{సాధన.}{$\mSat{M}{\lnot\Box\lnot !A}[w]$
      అనేదానికి సమానమైన షరతు
      $\mSat/{M}{\Box\lnot !A}[w]$.
      $\mSat{M}{\Box\lnot !A}[w]$
      వర్తించడానికి అవసరమైన, సరిపడే షరతు
      $Rww'$ ఉన్న ప్రతి $w'$లో
      $\mSat{M}{\lnot !A}[w']$ వర్తించడం.
      కాబట్టి $\mSat/{M}{\Box\lnot !A}[w]$
      వర్తించడానికి అవసరమైన, సరిపడే షరతు
      $Rww'$ ఉన్న ఏదో $w'$లో
      $\mSat/{M}{\lnot !A}[w']$ వర్తించడం.
      అలాగే $\mSat/{M}{\lnot !A}[w']$
      అనేది
      $\mSat{M}{!A}[w']$కు సమానం.
      అందువల్ల
      $\mSat{M}{\lnot\Box\lnot !A}[w]$
      వర్తించడానికి అవసరమైన, సరిపడే షరతు
      $Rww'$ ఉన్న ఏదో~$w'$లో
      $\mSat{M}{!A}[w']$ వర్తించడం.
      మళ్లీ $\mSat{M}{}[w]$
      నిర్వచనం ప్రకారం అది
      $\mSat{M}{\Diamond !A}[w]$కు
      సమానం.}
  \end{enumerate}
\end{proof}

\begin{probtag}{probDiamond}
  \olref[nml][syn][trw]{prop:dual}
  నిరూపణను పూర్తి చేయండి.
\end{probtag}
\end{tagblock}

\begin{prob}
  $\mModel{M} = \tuple{W, R, V}$
  ఒక నమూనా అనుకుందాం.
  $w_1, w_2 \in W$లకు:
  \begin{enumerate}
  \item ప్రతి
    \tetoken{ప్రతిజ్ఞావాక్య చరం}{!!{propositional variable}}
    $p$కు $w_1 \in V(p)$ కావడం, $w_2 \in V(p)$ కావడం సమానం; అలాగే
  \item ప్రతి $w \in W$కు:
    $Rw_1w$ కావడం, $Rw_2w$ కావడం సమానం.
  \end{enumerate}
  \tetoken{సూత్రాలపై}{!!{formula}s}
  ఆగమనంతో, ప్రతి
  \tetoken{సూత్రానికీ}{!!{formula}s}~$!A$కు
  $\mSat{M}{!A}[w_1]$ అనేది $\mSat{M}{!A}[w_2]$కు సమానం
  అని చూపండి.
\end{prob}

\begin{probtag}{prvDiamond}
  $\mModel{M} = \tuple{W, R, V}$
  అనుకుందాం. $\mSat{M}{\lnot\Diamond !A}[w]$
  అనేది
  $\mSat{M}{\Box\lnot!A}[w]$కు సమానం
  అని చూపండి.
\end{probtag}
