\olsection{ఒక నమూనాలో సత్యం}

కొన్నిసార్లు ఒక నమూనాలోని అన్ని లోకాల్లోనూ
సత్యమయ్యే \tetoken{సూత్రాలు}{!!{formula}s} ఏవో తెలుసుకోవాలనుకుంటాం.
దీనికి ఒక సంకేతాన్ని ప్రవేశపెడదాం.

\begin{defn}
  \tetoken{సూత్రం}{!!^a{formula}}~$!A$ అనేది $M = \tuple{W, R,
    V}$ అనే నమూనాలో \emph{సత్యం} అని $\mSat{M}{!A}$గా
  ఇది వర్తించడానికి అవసరమైన, సరిపడే షరతు ప్రతి $w \in W$కు $\mSat{M}{!A}[w]$
  వర్తించడం.
\end{defn}

\begin{prop}\ollabel{prop:truthfacts}
  \begin{enumerate}
  \item $\mSat{M}{!A}$ అయితే
    $\mSat/{M}{\lnot !A}$;
    \emph{తిరుగు దిశ వర్తించదు}.
  \item $\mSat{M}{!A \lif !B}$
    ఉంటే, $\mSat{M}{!A}$ నుంచి $\mSat{M}{!B}$ వస్తుంది.
    \emph{తిరుగు దిశ వర్తించదు}.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item $\mSat{M}{!A}$ అయితే
    $W$లోని అన్ని లోకాల్లో
    $!A$ సత్యం. $W \neq \emptyset$
    కాబట్టి $\mSat{M}{\lnot!A}$
    కావడం అసాధ్యం. అలా అయితే ఏదో
    లోకంలో $!A$ సత్యమూ,
    అసత్యమూ కావలసి వస్తుంది.

    ఇక తిరుగు దిశలో,
    $\mSat/{M}{\lnot !A}$
    అయితే ఏదో $w \in W$లో
    $!A$ సత్యం. అయితే
    \emph{ప్రతి}~$w \in W$కూ
    $\mSat{M}{!A}[w]$
    అని తేలదు. ఉదాహరణకు,
    \olref[rel]{fig:simple}
    నమూనాలో $\mSat/{M}{\lnot p}$,
    అలాగే $\mSat/{M}{p}$.
  \item $\mSat{M}{!A \lif !B}$
    మరియు $\mSat{M}{!A}$
    అనుకుందాం. $\mSat{M}{!B}$
    చూపడానికి, ఏదైనా
    $w \in W$ తీసుకుందాం.
    అప్పుడు $\mSat{M}{!A \lif !B}[w]$
    మరియు $\mSat{M}{!A}[w]$,
    కాబట్టి $\mSat{M}{!B}[w]$.
    $w$ ఏదైనా కావడంతో
    $\mSat{M}{!B}$.

    తిరుగు దిశ వర్తించదని
    చూపడానికి $\mSat{M}{!A}$
    అయితే $\mSat{M}{!B}$
    అనే మాట నిలిచినా
    $\mSat/{M}{!A \lif !B}$
    అయ్యే నమూనా
    $\mModel{M}$ కావాలి.
    \olref[rel]{fig:simple}
    నమూనానే మళ్లీ పరిశీలించండి.
    $\mSat/{M}{p}$ కాబట్టి
    $\mSat{M}{p}$ అయితే $\mSat{M}{q}$ అనే మాట
    శూన్యసందర్భంలో నిలుస్తుంది. కానీ
    $w_1$లో $p$ సత్యం,
    $q$ అసత్యం కనుక
    $\mSat/{M}{p \lif
      q}$.
  \end{enumerate}
\end{proof}

\begin{prob}
  $p_1$, $p_2$, $p_3$
  మాత్రమే \tetoken{ప్రతిజ్ఞావాక్య చరాలుగా}{!!{propositional variable}s}
  ఉన్న భాషకు కింది
  నమూనా $\mModel{M}$ను
  పరిశీలించండి:
  \begin{center}
    \begin{tikzpicture}[modal]
      \node[world] (w1) [label={[align=right]left:\mTrue{p_1}\\\mFalse{p_2}\\\mFalse{p_3}}]
            {$w_1$} ;
      \node[world] (w2) [label={[align=right]right:\mTrue{p_1}\\\mTrue{p_2}\\\mFalse{p_3}},
        below right=of w1]  {$w_2$};
      \node[world] (w3) [label={[align=right]right:\mTrue{p_1}\\\mTrue{p_2}\\\mTrue{p_3}},
        above right=of w2] {$w_3$};
      \draw[reflexive above] (w3) to (w3);
      \draw[->] (w1) to (w2);
      \draw[->] (w2) to (w3);
      \draw[->] (w1) to (w3);
    \end{tikzpicture}
  \end{center}
  కింది \tetoken{సూత్రాలు}{!!{formula}s},
  పథకాలు $\mModel{M}$లో
  సత్యమా, అంటే
  $\mModel{M}$లోని ప్రతి
  లోకంలో సత్యమా?
  మీ సమాధానాన్ని వివరించండి.
  \begin{enumerate}
  \item $p\lif \Diamond p$ ($p$ పరమాణు చరం);
  \item $!A\lif \Diamond !A$ ($!A$ ఏదైనా సూత్రం);
  \item $\Box p \lif p$ ($p$ పరమాణు చరం);
  \item $\lnot p \lif \Diamond \Box p$ ($p$ పరమాణు చరం);
  \item $\Diamond \Box !A$ ($!A$ ఏదైనా సూత్రం);
  \item $\Box \Diamond p$ ($p$ పరమాణు చరం).
  \end{enumerate}
\end{prob}
