\olsection{అదనపు నియమాల నిర్దుష్టత}

ఒక నమూనా వర్గంలోని ఏదో నమూనాలో \tetoken{ఒక టాబ్లో}{!!a{tableau}}
శాఖ సంతృప్తిపరచదగినదైనప్పుడల్లా, నియమం వర్తింపజేసిన
తర్వాతి శాఖ కూడా ఆ వర్గంలోని ఒక నమూనాలో
సంతృప్తిపరచదగినదైతే, ఆ నియమం ఆ నమూనా వర్గానికి
\emph{నిర్దుష్టమైనది} అంటాం.

\begin{prop}
  \ollabel{prop:soundness-T}
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ మరియు }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{స్వావర్తన నమూనాలకు నిర్దుష్టమైనది.}{స్వావర్తన నమూనాలకు నిర్దుష్టమైనవి.}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{$\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$కు
  T$\Box$ వర్తింపజేసి శాఖను విస్తరిస్తాం:
  \iftag{probBox}{వ్యాయామం.}{దీనివల్ల శాఖపై కొత్త
    \tetoken{చిహ్నిత సూత్రం}{!!{signed
      formula}}~$\sFmla{\True}{!B}[\sigma]$ వస్తుంది.
    $\mSat{M}{\Gamma}[f]$, ముఖ్యంగా
    $\mSat{M}{\Box !B}[f(\sigma)]$ అనుకుందాం. $R$
    స్వావర్తన సంబంధం కాబట్టి $Rf(\sigma)f(\sigma)$.
    అందువల్ల $\mSat{M}{!B}[f(\sigma)]$; అంటే~$\mModel{M}, f$,
    $\sFmla{\True}{!B}[\sigma]$ను సంతృప్తిపరుస్తుంది.}}{}
\tagitem{prvDiamond}{$\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$కు
  T$\Diamond$ వర్తింపజేసి శాఖను విస్తరిస్తాం:
  \iftag{probDiamond}{వ్యాయామం.}{దీనివల్ల శాఖపై కొత్త
    \tetoken{చిహ్నిత సూత్రం}{!!{signed
      formula}}~$\sFmla{\False}{!B}[\sigma]$ వస్తుంది.
    $\mSat{M}{\Gamma}[f]$, ముఖ్యంగా
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$ అనుకుందాం. $R$
    స్వావర్తన సంబంధం కాబట్టి $Rf(\sigma)f(\sigma)$.
    అందువల్ల $\mSat/{M}{!B}[f(\sigma)]$; అంటే~$\mModel{M}, f$,
    $\sFmla{\False}{!B}[\sigma]$ను సంతృప్తిపరుస్తుంది.}}{}
\end{tagenumerate}
\end{proof}

\end{tagenumerate}
\end{proof}
